스프레드시트에 데이터를 추가하면 출결 앱에 즉시 뜨고, 출결 앱에서 출석/결석/보강완료를 누르면 스프레드시트에 실시간으로 기록됩니다.

1 구글 스프레드시트 분석 완료

공유해주신 스프레드시트 구조(첫 번째 행 헤더 기준)에 맞춰 연동 스크립트를 재설계했습니다.
- 첫 번째 행의 열 구성(B열: 학년, C열: 이름, D열: 수업 시수, E~J열: 월~토 시작시간, K열: 특이사항, L열: 결석)을 완벽히 매핑합니다.
- 보강 관리를 위한 **AE열(보강일시)** 및 **AF열(보강완료)**은 아래 코드가 실행될 때 시트에 **자동으로 생성**되므로 기존 시트를 수정하실 필요가 없습니다.

2 구글 앱스 스크립트(Apps Script) 작성 및 배포

1. 구글 스프레드시트 상단 메뉴에서 [확장 프로그램] > [Apps Script]를 클릭합니다.
2. 기본 코드를 지우고 아래의 스크립트 코드를 복사하여 붙여넣습니다.
3. 상단의 [배포] > [새 배포]를 누르고 유형을 [웹 앱]으로 선택합니다.
4. 액세스 권한을 [모든 사용자(Anyone)]로 설정하여 배포한 후, 생성된 웹 앱 URL을 복사합니다.

function ensureColumns(sheet, requiredCols) {
  if (!sheet) return;
  var maxCols = sheet.getMaxColumns();
  if (maxCols < requiredCols) {
    sheet.insertColumnsAfter(maxCols, requiredCols - maxCols);
  }
}

function isSameStudentNameInGAS(name1, name2) {
  if (!name1 || !name2) return false;
  var c1 = String(name1).trim().replace(/\s+/g, "").toLowerCase();
  var c2 = String(name2).trim().replace(/\s+/g, "").toLowerCase();
  if (c1 === c2) return true;
  var pure1 = c1.replace(/^[\d\.\-\_]+/, "");
  var pure2 = c2.replace(/^[\d\.\-\_]+/, "");
  if (pure1 && pure2 && pure1 === pure2) return true;
  return false;
}

// 학년 순위를 계산하는 헬퍼 함수 (초1~6 -> 중1~3 -> 고1~3 순서 정렬용)
function getGradeRank(gradeStr) {
  if (!gradeStr) return 99;
  var match = gradeStr.match(/(초|중|고|대|유)?\s*(\d+)/);
  var type = match ? match[1] : "";
  var num = match ? parseInt(match[2], 10) : 0;
  
  var typeScore = 0;
  if (type === "초") typeScore = 10;
  else if (type === "중") typeScore = 20;
  else if (type === "고") typeScore = 30;
  else typeScore = 40;
  
  return typeScore + num;
}

// 결석일자 셀에 리치 텍스트 취소선(줄 긋기) 서식을 부분 적용하는 헬퍼 함수
function formatAbsentCell(cell, value) {
  if (!value) return;
  var text = String(value);
  var richTextBuilder = SpreadsheetApp.newRichTextValue().setText(text);
  
  var strikethroughStyle = SpreadsheetApp.newTextStyle().setStrikethrough(true).build();
  
  var regex = /~~([^~]+)~~/g;
  var match;
  while ((match = regex.exec(text)) !== null) {
    richTextBuilder.setTextStyle(match.index, regex.lastIndex, strikethroughStyle);
  }
  
  try {
    cell.setRichTextValue(richTextBuilder.build());
  } catch(e) {
    // Fallback if rich text fails
    cell.setValue(value);
  }
}

          var reasonStr = isAbsentToday ? "정규 결석" : (todayExtensionMinsVal > 0 ? "연장 (" + todayExtensionMinsVal + "분)" : "");
          dailyLogsToAdd.push({
            time: t, name: name, grade: grade, status: isAbsentToday ? "결석" : "대기", reason: reasonStr, notes: notes
          });
          regTimesAdded.push(t);
        }
      }
    }
    
    // 2. Makeup class time
    if (isMakeupToday) {
      if (!makeupTime) {
        makeupTime = "14:00"; // Fallback default time
      }
      
      // Self-healing check: if Column AF (makeupCompleted) is "완료", but they are NOT in today's daily log as completed,
      // it means this "완료" was left over from a previous makeup class. Reset it to "대기"!
      if (makeupCompleted === "완료" || makeupCompleted === "보강완료") {
        var alreadyCompletedToday = false;
        if (sheetDaily) {
          var lastRowDaily = sheetDaily.getLastRow();
          if (lastRowDaily >= 2) {
            var dailyData = sheetDaily.getRange(2, 1, lastRowDaily - 1, 6).getValues();
            for (var dIdx = 0; dIdx < dailyData.length; dIdx++) {
              var dDate = String(dailyData[dIdx][0]).trim();
              var dName = String(dailyData[dIdx][2]).trim();
              var dStatus = String(dailyData[dIdx][4]).trim();
              if (dName === name && isSameDateGAS(dDate, m + "/" + d + dayShort) && (dStatus === "수업완료" || dStatus === "보강완료")) {
                alreadyCompletedToday = true;
                break;
              }
            }
          }
        }
        if (!alreadyCompletedToday) {
          makeupCompleted = "보강대기";
          sheetStudents.getRange(i + 2, 32).setValue("보강대기"); // Reset Column AF to "보강대기"
        }
      }
      
      if (makeupCompleted !== "완료" && makeupCompleted !== "보강완료") {
        var isDuplicateTime = regTimesAdded.some(function(rt) {
          return rt.replace(/\s+/g, "") === makeupTime.replace(/\s+/g, "");
        });
        if (!isDuplicateTime) {
          dailyLogsToAdd.push({
            time: makeupTime, name: name, grade: grade, status: "대기", reason: "보강 수업", notes: notes
          });
        }
      }
    }
  }
  
  dailyLogsToAdd.sort(function(a, b) {
    var timeComp = a.time.localeCompare(b.time);
    if (timeComp !== 0) return timeComp;
    var rankA = getGradeRank(a.grade);
    var rankB = getGradeRank(b.grade);
    if (rankA !== rankB) return rankA - rankB;
    return a.name.localeCompare(b.name);
  });
  
  if (dailyLogsToAdd.length > 0) {
    // 3행부터 밀어넣기 시작
    var writeRange = sheetDaily.getRange(3, 1, dailyLogsToAdd.length, 11);
    var writeValues = [];
    for (var j = 0; j < dailyLogsToAdd.length; j++) {
      writeValues.push([
        dateStr, dailyLogsToAdd[j].time, dailyLogsToAdd[j].name, dailyLogsToAdd[j].notes,
        dailyLogsToAdd[j].status, "", dailyLogsToAdd[j].reason, "", "", "", ""
      ]);
    }
    writeRange.setValues(writeValues);
    
    // 유효성 룰 전파 (3행 이하)
    if (validationE2) sheetDaily.getRange(3, 5, dailyLogsToAdd.length, 1).setDataValidation(validationE2);
    if (validationG2) sheetDaily.getRange(3, 7, dailyLogsToAdd.length, 1).setDataValidation(validationG2);
    if (validationH2) sheetDaily.getRange(3, 8, dailyLogsToAdd.length, 1).setDataValidation(validationH2);
    if (validationI2) sheetDaily.getRange(3, 9, dailyLogsToAdd.length, 1).setDataValidation(validationI2);
  }
}

// 전체시간표를 기반으로 요일별/시간대별/학년별(초등저, 초등고, 중등, 한국사) visual 오늘 시간표 보드를 자동 생성하는 함수
function generateWeeklyTimetable() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetStudents = ss.getSheetByName("전체시간표") || ss.getSheetByName("전체 시간표");
  if (!sheetStudents) return;
  var lastRowSt = sheetStudents.getLastRow();
  if (lastRowSt < 2) return;
  
  var studentsData = sheetStudents.getRange(2, 1, lastRowSt - 1, 32).getDisplayValues();
  
  var sheetWeekly = ss.getSheetByName("오늘 시간표");
  if (sheetWeekly) {
    sheetWeekly.clear();
    sheetWeekly.clearFormats();
  } else {
    sheetWeekly = ss.insertSheet("오늘 시간표");
  }
  sheetWeekly.setHiddenGridlines(false);
  
  var WEEKDAYS = ['월요일', '화요일', '수요일', '목요일', '금요일', '토요일'];
  var CATEGORIES = ['초등저', '초등고', '중등', '한국사'];
  
  var timetableMap = {};
  var timesSet = {};
  
  var standardSlots = ["13:00", "13:30", "14:00", "14:30", "15:00", "15:30", "16:00", "16:30", "17:00", "17:30", "18:00", "18:30", "19:00", "19:30", "20:00", "20:30"];
  standardSlots.forEach(function(slot) {
    timesSet[slot] = true;
  });
  
  function normalizeTimeSlot(timeStr) {
    if (!timeStr) return null;
    var match = timeStr.match(/(\d{1,2})\s*:\s*(\d{2})/);
    if (!match) return null;
    var hr = parseInt(match[1], 10);
    var min = parseInt(match[2], 10);
    if (isNaN(hr) || isNaN(min)) return null;
    if (min >= 0 && min < 30) {
      min = 0;
    } else {
      min = 30;
    }
    return String(hr).padStart(2, '0') + ":" + String(min).padStart(2, '0');
  }
  
  function getCategory(grade, classes, name) {
    var isHistory = (classes.indexOf("한국사") !== -1 || classes.indexOf("세계사") !== -1 || classes.indexOf("역사") !== -1 || name.indexOf("한국사") !== -1 || name.indexOf("세계사") !== -1 || name.indexOf("역사") !== -1);
    if (isHistory) return "한국사";
    
    if (grade.indexOf("초1") !== -1 || grade.indexOf("초2") !== -1 || grade.indexOf("초3") !== -1 || grade.indexOf("1학년") !== -1 || grade.indexOf("2학년") !== -1 || grade.indexOf("3학년") !== -1) {
      return "초등저";
    }
    if (grade.indexOf("초4") !== -1 || grade.indexOf("초5") !== -1 || grade.indexOf("초6") !== -1 || grade.indexOf("4학년") !== -1 || grade.indexOf("5학년") !== -1 || grade.indexOf("6학년") !== -1) {
      return "초등고";
    }
    if (grade.indexOf("중") !== -1 || grade.indexOf("고") !== -1) {
      return "중등";
    }
    var numMatch = grade.match(/\d+/);
    if (numMatch) {
      var num = parseInt(numMatch[0], 10);
      if (num <= 3) return "초등저";
      if (num <= 6) return "초등고";
    }
    return "초등저";
  }
  
  for (var i = 0; i < studentsData.length; i++) {
    var name = String(studentsData[i][2] || "").trim();
    var grade = String(studentsData[i][1] || "").trim();
    var classes = String(studentsData[i][3] || "").trim();
    if (!name) continue;
    
    var keywords = ["국어", "세계사", "한국사", "독서", "문해력", "내신", "초등", "중등", "특강"];
    var isGroup = false;
    for (var k = 0; k < keywords.length; k++) {
      if (name.indexOf(keywords[k]) !== -1) {
        isGroup = true;
        break;
      }
    }
    if (isGroup || name.length > 8) continue;
    
    var cat = getCategory(grade, classes, name);
    
    var dayCols = { "월요일": 4, "화요일": 5, "수요일": 6, "목요일": 7, "금요일": 8, "토요일": 9 };
    for (var dName in dayCols) {
      var timeCell = String(studentsData[i][dayCols[dName]] || "").trim();
      if (!timeCell) continue;
      
      var timeParts = timeCell.split(/[,/; ]+/);
      for (var tp = 0; tp < timeParts.length; tp++) {
        var rawTime = timeParts[tp].trim();
        if (!rawTime) continue;
        
        var normalized = normalizeTimeSlot(rawTime);
        if (normalized) {
          timesSet[normalized] = true;
          
          var key = dName + "_" + normalized + "_" + cat;
          if (!timetableMap[key]) {
            timetableMap[key] = [];
          }
          timetableMap[key].push({
            name: name,
            grade: grade,
            classes: classes
          });
        }
      }
    }
  }
  
  // 요일별/시간대별/카테고리별 학생 목록을 학년 순 -> 이름 순으로 정렬
  for (var key in timetableMap) {
    timetableMap[key].sort(function(a, b) {
      var rankA = getGradeRank(a.grade);
      var rankB = getGradeRank(b.grade);
      if (rankA !== rankB) return rankA - rankB;
      return a.name.localeCompare(b.name);
    });
  }
  
  var sortedTimes = Object.keys(timesSet).sort();
  var grid = [];
  
  var titleRow = [];
  for (var c = 0; c < 30; c++) {
    titleRow.push(c === 0 ? "♣ 오늘 시간표 (학생 정원 파악 보드) - 정원 12명 제한" : "");
  }
  grid.push(titleRow);
  
  var dayRow = ["시간"];
  WEEKDAYS.forEach(function(day) {
    dayRow.push(day);
    dayRow.push("");
    dayRow.push("");
    dayRow.push("");
    dayRow.push("");
  });
  dayRow.pop();
  grid.push(dayRow);
  
  var catRow = [""];
  WEEKDAYS.forEach(function(day) {
    catRow.push("초등저");
    catRow.push("초등고");
    catRow.push("중등");
    catRow.push("한국사");
    catRow.push("");
  });
  catRow.pop();
  grid.push(catRow);
  
  var dayOffsets = { "월요일": 1, "화요일": 6, "수요일": 11, "목요일": 16, "금요일": 21, "토요일": 26 };
  var catOffsets = { "초등저": 0, "초등고": 1, "중등": 2, "한국사": 3 };
  
  var mergesToApply = [];
  var cellColorStyles = []; // elements: {row: 1-based, col: 1-based, bg: ..., fg: ...}
  var warningStyles = []; // elements: {row: 1-based, numRows: ..., val: ...}
  
  var currentRowIdx = 4; // 1-based row index for grid rows
  
  sortedTimes.forEach(function(timeSlot) {
    // 1. 해당 시간대의 모든 요일/카테고리 중 최대 학생 수 구하기 (그만큼 행 생성)
    var maxRowsForSlot = 1;
    WEEKDAYS.forEach(function(day) {
      CATEGORIES.forEach(function(cat) {
        var key = day + "_" + timeSlot + "_" + cat;
        var students = timetableMap[key] || [];
        if (students.length > maxRowsForSlot) {
          maxRowsForSlot = students.length;
        }
      });
    });
    
    // 2. 최대 학생 수만큼 행(Row) 생성하여 셀마다 학생 1명씩 배치
    for (var r = 0; r < maxRowsForSlot; r++) {
      var row = [r === 0 ? timeSlot : ""];
      for (var c = 1; c < 30; c++) {
        row.push("");
      }
      
      WEEKDAYS.forEach(function(day) {
        var dayOffset = dayOffsets[day];
        CATEGORIES.forEach(function(cat) {
          var key = day + "_" + timeSlot + "_" + cat;
          var students = timetableMap[key] || [];
          if (students[r]) {
            var s = students[r];
            row[dayOffset + catOffsets[cat]] = s.name + "(" + s.grade + ")";
            
            // 수강과목 키워드별 파스텔 컬러 지정 (한국사/문해력/내신/특강 등)
            var classesStr = String(s.classes || "").toLowerCase();
            var nameStr = String(s.name || "").toLowerCase();
            var bg = "#ecfdf5"; // 기본 연녹색 (일반)
            var fg = "#065f46";
            
            if (cat === "한국사" || classesStr.indexOf("한국사") !== -1 || classesStr.indexOf("세계사") !== -1 || classesStr.indexOf("역사") !== -1 || nameStr.indexOf("한국사") !== -1 || nameStr.indexOf("세계사") !== -1 || nameStr.indexOf("역사") !== -1) {
              bg = "#fef3c7"; // 연주황 (역사)
              fg = "#92400e";
            } else if (classesStr.indexOf("문해력") !== -1 || classesStr.indexOf("독서") !== -1 || classesStr.indexOf("국어") !== -1 || nameStr.indexOf("문해력") !== -1 || nameStr.indexOf("독서") !== -1 || nameStr.indexOf("국어") !== -1) {
              bg = "#dbeafe"; // 연파랑 (문해력)
              fg = "#1e40af";
            } else if (classesStr.indexOf("내신") !== -1 || classesStr.indexOf("시험") !== -1 || nameStr.indexOf("내신") !== -1 || nameStr.indexOf("시험") !== -1) {
              bg = "#f3e8ff"; // 연보라 (내신)
              fg = "#6b21a8";
            } else if (classesStr.indexOf("특강") !== -1 || nameStr.indexOf("특강") !== -1) {
              bg = "#fce7f3"; // 연분홍 (특강)
              fg = "#9d174d";
            }
            
            cellColorStyles.push({
              row: currentRowIdx,
              col: dayOffset + catOffsets[cat] + 1,
              bg: bg,
              fg: fg
            });
          }
        });
      });
      grid.push(row);
      currentRowIdx++;
    }
    
    // 시간 컬럼 A열 병합 정보 기록
    if (maxRowsForSlot > 1) {
      mergesToApply.push({
        row: currentRowIdx - maxRowsForSlot,
        col: 1,
        numRows: maxRowsForSlot,
        numCols: 1
      });
    }
    
    // 3. 정원 초과 계산 (한 시간대 당일 총원 > 12)
    var warnings = [];
    var exceedDayIndices = [];
    WEEKDAYS.forEach(function(day, dayIdx) {
      var totalInSlot = 0;
      CATEGORIES.forEach(function(cat) {
        var key = day + "_" + timeSlot + "_" + cat;
        var students = timetableMap[key] || [];
        totalInSlot += students.length;
      });
      
      if (totalInSlot > 12) {
        warnings.push(day.substring(0, 1) + ":" + totalInSlot + "명");
        exceedDayIndices.push(dayIdx);
      }
    });
    
    if (warnings.length > 0) {
      var warningRow = currentRowIdx - maxRowsForSlot;
      warningStyles.push({
        row: warningRow,
        numRows: maxRowsForSlot,
        val: timeSlot + " (" + warnings.join(", ") + ")"
      });
      
      // 초과한 요일의 4개 열 전체 행 영역을 연한 빨간색 경고 색상으로 오버라이드
      exceedDayIndices.forEach(function(dayIdx) {
        var startCol = 2 + dayIdx * 5; // 1-based col
        for (var rIdx = currentRowIdx - maxRowsForSlot; rIdx < currentRowIdx; rIdx++) {
          for (var colIdx = startCol; colIdx < startCol + 4; colIdx++) {
            cellColorStyles.push({
              row: rIdx,
              col: colIdx,
              bg: "#ffe4e6", // 경고용 연한 빨간색
              fg: null // 글자색은 학생 과목 고유색을 유지할 수 있도록 null 처리
            });
          }
        }
      });
    }
  });
  
  // 전체 범위에 그리드 값 채우기
  var range = sheetWeekly.getRange(1, 1, grid.length, 30);
  range.setValues(grid);
  
  // 1. 헤더/기본 스타일 채우기
  sheetWeekly.getRange(1, 1, 1, 30).merge().setFontSize(13).setFontWeight("bold").setBackground("#f1f5f9").setFontColor("#1e293b").setHorizontalAlignment("left").setWrap(false);
  
  WEEKDAYS.forEach(function(day, index) {
    var col = 2 + index * 5;
    sheetWeekly.getRange(2, col, 1, 4).merge().setFontWeight("bold").setHorizontalAlignment("center");
  });
  
  var pastelColors = ["#e0f2fe", "#f0fdf4", "#fff7ed", "#fdf2f8", "#ecfdf5", "#f5f3ff"];
  WEEKDAYS.forEach(function(day, index) {
    var col = 2 + index * 5;
    var color = pastelColors[index % pastelColors.length];
    
    sheetWeekly.getRange(2, col, 1, 4).setBackground(color).setFontColor("#0f172a");
    sheetWeekly.getRange(3, col, 1, 4).setBackground(color).setFontColor("#334155").setFontStyle("italic").setFontSize(9).setHorizontalAlignment("center");
  });
  
  // 2. 바디 영역 벌크 색상 배열 초기화 및 적용 (Apps Script 실행 속도 극대화)
  var totalBodyRows = grid.length - 3;
  var backgrounds = [];
  var fontColors = [];
  
  for (var r = 0; r < totalBodyRows; r++) {
    var bgRow = [];
    var fgRow = [];
    for (var c = 0; c < 30; c++) {
      if (c === 0) {
        bgRow.push("#f8fafc"); // 시간 컬럼
        fgRow.push("#64748b");
      } else if (c % 5 === 0) {
        bgRow.push("#f1f5f9"); // 구분 스페이서 열
        fgRow.push("#64748b");
      } else {
        bgRow.push("#ffffff"); // 빈 셀
        fgRow.push("#334155");
      }
    }
    backgrounds.push(bgRow);
    fontColors.push(fgRow);
  }
  
  // 3. 기록된 학생 개별 셀 및 경고 셀 스타일 덮어쓰기
  cellColorStyles.forEach(function(style) {
    var rIdx = style.row - 4; // 0-indexed body row
    var cIdx = style.col - 1; // 0-indexed column
    if (rIdx >= 0 && rIdx < totalBodyRows && cIdx >= 0 && cIdx < 30) {
      if (style.bg) backgrounds[rIdx][cIdx] = style.bg;
      if (style.fg) fontColors[rIdx][cIdx] = style.fg;
    }
  });
  
  // 경고 표시 셀 A열 오버라이드
  warningStyles.forEach(function(w) {
    var rIdx = w.row - 4;
    if (rIdx >= 0 && rIdx < totalBodyRows) {
      for (var k = 0; k < w.numRows; k++) {
        backgrounds[rIdx + k][0] = "#fee2e2";
        fontColors[rIdx + k][0] = "#991b1b";
      }
    }
  });
  
  // 벌크로 시트 서식 적용 (단 1번의 API 호출)
  var bodyRange = sheetWeekly.getRange(4, 1, totalBodyRows, 30);
  bodyRange.setBackgrounds(backgrounds);
  bodyRange.setFontColors(fontColors);
  
  // 4. 시간 컬럼 병합 및 경고 텍스트 주입
  mergesToApply.forEach(function(m) {
    try {
      sheetWeekly.getRange(m.row, m.col, m.numRows, m.numCols).merge();
    } catch(e) {}
  });
  
  warningStyles.forEach(function(w) {
    sheetWeekly.getRange(w.row, 1).setValue(w.val);
  });
  
  // 5. 열 넓이 및 정렬 서식 마무리
  for (var index = 0; index < 5; index++) {
    var col = 6 + index * 5;
    sheetWeekly.setColumnWidth(col, 15);
  }
  
  sheetWeekly.setColumnWidth(1, 140);
  for (var index = 0; index < 6; index++) {
    var colStart = 2 + index * 5;
    for (var cOffset = 0; cOffset < 4; cOffset++) {
      sheetWeekly.setColumnWidth(colStart + cOffset, 95);
    }
  }
  
  // 정렬 및 테두리선
  sheetWeekly.getRange(4, 1, totalBodyRows, 1).setVerticalAlignment("middle").setHorizontalAlignment("center").setFontWeight("bold");
  var contentRange = sheetWeekly.getRange(4, 2, totalBodyRows, 29);
  contentRange.setWrap(true).setVerticalAlignment("middle").setHorizontalAlignment("center").setFontSize(9);
  
  sheetWeekly.getRange(2, 1, grid.length - 1, 30).setBorder(true, true, true, true, true, true, "#e2e8f0", SpreadsheetApp.BorderStyle.SOLID);
}

// 최초 1회 실행하여 필요한 시트 탭들과 헤더 열 구성을 자동으로 생성해 주는 함수
function setupSheetsAndHeaders() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var sheetStudents = ss.getSheetByName("전체시간표") || ss.getSheetByName("전체 시간표");
  if (sheetStudents) {
    ensureColumns(sheetStudents, 36);
    if (sheetStudents.getRange("AE1").getValue() !== "보강일시") sheetStudents.getRange("AE1").setValue("보강일시");
    if (sheetStudents.getRange("AF1").getValue() !== "보강완료") sheetStudents.getRange("AF1").setValue("보강완료");
    if (sheetStudents.getRange("AG1").getValue() !== "결석 누적(분)") sheetStudents.getRange("AG1").setValue("결석 누적(분)");
    if (sheetStudents.getRange("AH1").getValue() !== "보강 완료(분)") sheetStudents.getRange("AH1").setValue("보강 완료(분)");
    if (sheetStudents.getRange("AI1").getValue() !== "보강 잔여(분)") sheetStudents.getRange("AI1").setValue("보강 잔여(분)");
    if (sheetStudents.getRange("AJ1").getValue() !== "오늘 연장(분)") sheetStudents.getRange("AJ1").setValue("오늘 연장(분)");
    
    var lastRowSt = sheetStudents.getLastRow();
    if (lastRowSt >= 2) {
      sheetStudents.getRange(2, 35, lastRowSt - 1, 1).setFormulaR1C1("=RC[-2]-RC[-1]");
    }
  }
  
  // 출결보강관리 셋업

// 6열 고정 넓이 달력 그리기용 출결 달력 검색 시트 생성 및 그리기 함수
function updateAttendanceCalendarSheet(sheet, studentName, year, month) {
  if (!sheet) return;
  
  // 1. 5행 이하 전체 삭제 및 초기화
  var lastRow = sheet.getLastRow();
  if (lastRow >= 5) {
    sheet.getRange(5, 1, lastRow - 4, 7).clear();
    sheet.getRange(5, 1, lastRow - 4, 7).clearFormat();
  }
  
  // 2. 4행 요일 헤더 설정
  sheet.getRange("A4:G4").setValues([["일", "월", "화", "수", "목", "금", "토"]])
       .setFontWeight("bold")
       .setBackground("#e2e8f0")
       .setHorizontalAlignment("center")
       .setVerticalAlignment("middle");
       
  for (var c = 1; c <= 7; c++) {
    sheet.setColumnWidth(c, 110);
  }
  
  // 3. 학생 데이터 수집
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetDaily = ss.getSheetByName("출결보강관리") || ss.getSheetByName("오늘 출석부");
  var sheetAccum = ss.getSheetByName("학생수업관리") || ss.getSheetByName("누적일일수업관리");
  
  var logs = [];
  
  // 오늘 출석부에서 수집
  if (sheetDaily) {
    var lastDaily = sheetDaily.getLastRow();
    if (lastDaily > 2) {
      var dailyData = sheetDaily.getRange(3, 1, lastDaily - 2, 11).getDisplayValues();
      for (var i = 0; i < dailyData.length; i++) {
        var rowName = String(dailyData[i][2] || "").trim();
        if (rowName.replace(/\s+/g, '') === studentName.replace(/\s+/g, '')) {
          logs.push({
            date: String(dailyData[i][0] || "").trim(),
            time: String(dailyData[i][1] || "").trim(),
            status: String(dailyData[i][4] || "").trim(),
            inTime: String(dailyData[i][5] || "").trim()
          });
        }
      }
    }
  }
  
  // 누적일일수업관리에서 수집
  if (sheetAccum) {
    var lastAccum = sheetAccum.getLastRow();
    if (lastAccum > 1) {
      // 최근 3000행만 조회 (로딩 속도 보존)
      var startRow = Math.max(2, lastAccum - 3000);
      var numRows = lastAccum - startRow + 1;
      var accumData = sheetAccum.getRange(startRow, 1, numRows, 11).getDisplayValues();
      for (var i = 0; i < accumData.length; i++) {
        var rowName = String(accumData[i][2] || "").trim();
        if (rowName.replace(/\s+/g, '') === studentName.replace(/\s+/g, '')) {
          logs.push({
            date: String(accumData[i][0] || "").trim(),
            time: String(accumData[i][1] || "").trim(),
            status: String(accumData[i][4] || "").trim(),
            inTime: String(accumData[i][5] || "").trim()
          });
        }
      }
    }
  }
  
  // 4. 달력 일자 루프 및 채우기
  var firstDay = new Date(year, month - 1, 1);
  var numDays = new Date(year, month, 0).getDate();
  var startDayOfWeek = firstDay.getDay(); // 0: 일요일, 6: 토요일
  
  var gridRow = 5;
  var gridCol = startDayOfWeek + 1;
  
  // 달력 행 높이 설정 (정사각형 느낌)
  for (var r = 5; r <= 11; r++) {
    sheet.setRowHeight(r, 65);
  }
  
  // 이전 달 빈 칸 연한 회색 배경 처리
  if (startDayOfWeek > 0) {
    sheet.getRange(5, 1, 1, startDayOfWeek).setBackground("#f8fafc");
  }
  
  for (var day = 1; day <= numDays; day++) {
    var cell = sheet.getRange(gridRow, gridCol);
    var cellText = day + "\n";
    
    // 이 날짜의 출결 로그 검색 (예: "8/9" 또는 "8/9일" 매칭)
    var matchingLog = null;
    var datePattern = new RegExp('^' + month + '[/\\.]' + day + '(\\D|$)');
    
    for (var l = 0; l < logs.length; l++) {
      if (datePattern.test(logs[l].date)) {
        matchingLog = logs[l];
        break;
      }
    }
    
    if (matchingLog) {
      var status = matchingLog.status;
      var inTime = matchingLog.inTime || matchingLog.time || "";
      if (status === "대기" || status === "수업중" || status === "수업 중" || status === "보강 수업중") {
        cellText += status;
        cell.setBackground("#f1f5f9").setFontColor("#475569");
      } else if (status === "출석" || status === "수업완료") {
        cellText += "출석 (" + inTime + ")";
        cell.setBackground("#dcfce7").setFontColor("#15803d");
      } else if (status === "결석") {
        cellText += "결석";
        cell.setBackground("#fee2e2").setFontColor("#b91c1c");
      } else if (status === "보강대기") {
        cellText += "보강대기";
        cell.setBackground("#fce7f3").setFontColor("#be185d");
      } else if (status === "보강완료" || status === "완료") {
        cellText += "보강완료 (" + inTime + ")";
        cell.setBackground("#e0f2fe").setFontColor("#0369a1");
      } else if (status === "휴강") {
        cellText += "휴강";
        cell.setBackground("#f8fafc").setFontColor("#94a3b8");
      } else {
        cellText += status;
      }
    }
    
    cell.setValue(cellText)
        .setFontSize(9)
        .setFontFamily("Malgun Gothic")
        .setHorizontalAlignment("center")
        .setVerticalAlignment("top");
        
    gridCol++;
    if (gridCol > 7) {
      gridCol = 1;
      gridRow++;
    }
  }
  
  // 다음 달 빈 칸 연한 회색 배경 처리
  if (gridCol <= 7) {
    sheet.getRange(gridRow, gridCol, 1, 8 - gridCol).setBackground("#f8fafc");
  }
  if (gridRow < 10) {
    sheet.getRange(gridRow + 1, 1, 10 - gridRow, 7).setBackground("#f8fafc");
  }
  
  // 전체 테두리선 추가
  sheet.getRange(5, 1, gridRow - 5 + 1, 7).setBorder(true, true, true, true, true, true, "#e2e8f0", SpreadsheetApp.BorderStyle.SOLID);
}

// 출결보강관리의 숫자(H열), 이벤트(I열), 진단/글쓰기(J열) 또는 수업내용(K열) 입력 시 회원 분석 시트로 자동 누적 기록하는 함수
function accumulateDailyEventToMemberAnalysis(sheetDaily, rowNumber) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetMember = getMemberAnalysisSheet(ss);
  if (!sheetMember) return;

  var rowValues = sheetDaily.getRange(rowNumber, 1, 1, 11).getValues()[0];
  var dateStr = String(rowValues[0] || "").trim(); // Col A
  var name = String(rowValues[2] || "").trim(); // Col C
  var numVal = String(rowValues[7] || "").trim(); // Col H
  var eventVal = String(rowValues[8] || "").trim(); // Col I
  var grammarDoneVal = String(rowValues[9] || "").trim(); // Col J (진단글쓰기/특별수업)
  var contentsVal = String(rowValues[10] || "").trim(); // Col K (수업내용 및 특이사항)

  if (!name) return;

  // Auto-populate today's date in Col A if empty
  if (!dateStr && (numVal || eventVal || grammarDoneVal || contentsVal)) {
    var today = new Date();
    var weekdays = ["일", "월", "화", "수", "목", "금", "토"];
    var todayDay = weekdays[today.getDay()];
    dateStr = (today.getMonth() + 1) + "/" + today.getDate() + todayDay;
    sheetDaily.getRange(rowNumber, 1).setValue(dateStr);
  }

  var memberLastRow = sheetMember.getLastRow();
  if (memberLastRow < 2) return;
  var memberNames = sheetMember.getRange(2, 2, memberLastRow - 1, 1).getValues();
  var memberRowIndex = -1;
  var targetCleanName = name.replace(/\s+/g, '');
  for (var i = 0; i < memberNames.length; i++) {
    var cleanMemberName = String(memberNames[i][0] || '').replace(/\s+/g, '');
    if (cleanMemberName && cleanMemberName === targetCleanName) {
      memberRowIndex = 2 + i;
      break;
    }
  }

  if (memberRowIndex === -1) return;

  var mdMatch = dateStr.match(/(\d+\/\d+)/);
  var dateFormatted = "";
  if (mdMatch) {
    dateFormatted = mdMatch[1];
  } else {
    var today = new Date();
    dateFormatted = (today.getMonth() + 1) + "/" + today.getDate();
  }
  
  var yy = "26";
  try {
    var yearVal = new Date().getFullYear();
    yy = String(yearVal).substring(2);
  } catch(err) {}
  var eventDatePrefix = yy + "." + dateFormatted;

  // Parse multi-line / comma-separated / semicolon-separated entries
  var splitRegex = /[\n,;]+/;
  var numLines = numVal ? numVal.split(splitRegex).map(function(s) { return s.trim(); }).filter(Boolean) : [];
  var eventLines = eventVal ? eventVal.split(splitRegex).map(function(s) { return s.trim(); }).filter(Boolean) : [];
  var grammarLines = grammarDoneVal ? grammarDoneVal.split(splitRegex).map(function(s) { return s.trim(); }).filter(Boolean) : [];

  var maxLen = Math.max(numLines.length, eventLines.length);

  var groupMap = {
    7: [],
    8: [],
    10: [],
    11: [],
    12: []
  };

  // [핵심 개편] F열(6열: 코칭 방향 & 주력 목표)에는 더 이상 일일 수업내용(K열)을 누적하지 않습니다.
  // 일일 수업 활동(숙제현황·성실발자국·독서발자국·문해력발자국·나의기록)은 18열 [학습기록] 시트에 즉시 동기화합니다!
  if (contentsVal) {
    try {
      var fpNow = parseDailyContentsFootprints_(contentsVal);
      if (fpNow) {
        var sheetRecNow = vocabEnsureRecordSheet_(ss);
        var dParts = dateFormatted.split("/");
        var isoDateNow = vocabFmtDate_(new Date());
        if (dParts.length === 2) {
          var mmNow = parseInt(dParts[0], 10);
          var ddNow = parseInt(dParts[1], 10);
          var yyyyNow = new Date().getFullYear();
          if (!isNaN(mmNow) && !isNaN(ddNow)) {
            isoDateNow = yyyyNow + "-" + (mmNow < 10 ? "0" + mmNow : mmNow) + "-" + (ddNow < 10 ? "0" + ddNow : ddNow);
          }
        }
        vocabUpsertStudentDayRecords_(sheetRecNow, targetCleanName, [{
          date: isoDateNow,
          type: "일일요약",
          hwStatus: fpNow.hwStatus,
          step1Memo: fpNow.step1Memo,
          step2Memo: fpNow.step2Memo,
          literacyMemo: fpNow.literacyMemo,
          reflection: fpNow.reflection
        }]);
      }
    } catch (fpErr) {}
  }

  for (var idx = 0; idx < maxLen; idx++) {
    var num = numLines[idx] || "";
    var ev = eventLines[idx] || "";
    if (!num && !ev) continue;

    var textToAppend = eventDatePrefix;
    if (num) textToAppend += " " + num;
    if (ev) textToAppend += " " + ev;

    var targetCol = 8; // Default Col H
    var lowerEvent = ev.toLowerCase();
    
    if (lowerEvent.indexOf("코스업") !== -1 || lowerEvent.indexOf("레벨업") !== -1 || lowerEvent.indexOf("진로검사") !== -1) {
      targetCol = 8;
    } else if (lowerEvent.indexOf("진단") !== -1 || lowerEvent.indexOf("진도") !== -1) {
      targetCol = 7;
    } else if (lowerEvent.indexOf("훈민정음") !== -1 || lowerEvent.indexOf("한자") !== -1 || lowerEvent.indexOf("교재") !== -1) {
      targetCol = 10;
    } else if (lowerEvent.indexOf("독서계획표") !== -1 || lowerEvent.indexOf("친구") !== -1 || lowerEvent.indexOf("탐험가") !== -1 || lowerEvent.indexOf("고수") !== -1 || lowerEvent.indexOf("챔피언") !== -1 || lowerEvent.indexOf("마스터") !== -1 || lowerEvent.indexOf("달성") !== -1 || lowerEvent.indexOf("스티커") !== -1) {
      targetCol = 12;
    } else if (lowerEvent.indexOf("읽트") !== -1 || lowerEvent.indexOf("독트") !== -1 || lowerEvent.indexOf("읽기") !== -1 || lowerEvent.indexOf("독서") !== -1) {
      targetCol = 11;
    } else {
      targetCol = 8;
    }

    groupMap[targetCol].push(textToAppend);
  }

  // Parse Column J (grammarDone) items
  for (var g = 0; g < grammarLines.length; g++) {
    var item = grammarLines[g];
    var isCompletion = (item.indexOf("글완") !== -1 || item.indexOf("글쓰기완") !== -1 || item.indexOf("완료") !== -1);
    var hasDigit = /\d+/.test(item);

    if (isCompletion) {
      // Find existing book line in progress (Col G / Column 7) of memberAnalysis and append " 글완"
      var cleanBookAndScore = item.replace(/글완|글쓰기완료|글쓰기완|완료/g, "").trim();
      var cleanBook = cleanBookAndScore.replace(/\s*\d+점?$/, "").trim();
      var cleanBookNoSpaces = cleanBook.replace(/\s+/g, "").toLowerCase();

      var progressCell = sheetMember.getRange(memberRowIndex, 7);
      var currentProgress = String(progressCell.getValue() || "").trim();
      if (currentProgress) {
        var lines = currentProgress.split("\n");
        var found = false;
        for (var lIdx = 0; lIdx < lines.length; lIdx++) {
          var lineNoSpaces = lines[lIdx].replace(/\s+/g, "").toLowerCase();
          if (cleanBookNoSpaces && lineNoSpaces.indexOf(cleanBookNoSpaces) !== -1) {
            found = true;
            var newLine = lines[lIdx];
            if (/\d+/.test(cleanBookAndScore)) {
              newLine = eventDatePrefix + " " + cleanBookAndScore + " 글완";
            } else {
              if (lines[lIdx].indexOf("글완") === -1 && lines[lIdx].indexOf("글쓰기완") === -1 && lines[lIdx].indexOf("완료") === -1) {
                newLine = lines[lIdx] + " 글완";
              }
            }
            lines[lIdx] = newLine;
            if (newLine.trim().indexOf(eventDatePrefix) === 0) {
              groupMap[7].push(newLine);
            }
          }
        }
        if (!found && cleanBookNoSpaces) {
          var newLine = eventDatePrefix + " " + cleanBookAndScore + " 글완";
          lines.push(newLine);
          groupMap[7].push(newLine);
          found = true;
        } else if (!found && !cleanBookNoSpaces && lines.length > 0) {
          var lastLine = lines[lines.length - 1];
          if (lastLine.indexOf("글완") === -1 && lastLine.indexOf("글쓰기완") === -1 && lastLine.indexOf("완료") === -1) {
            var newLine = lastLine + " 글완";
            if (newLine.trim().indexOf(eventDatePrefix) === 0) {
              groupMap[7].push(newLine);
            }
            lines[lines.length - 1] = newLine;
          }
        }
        progressCell.setValue(lines.join("\n"));
      } else {
        var textToSet = eventDatePrefix + " " + (cleanBookAndScore ? cleanBookAndScore + " 글완" : "글완");
        progressCell.setValue(textToSet);
        groupMap[7].push(textToSet);
      }
    } else if (hasDigit) {
      var datePattern = /^\d+[\./]\d+/;
      var textToAppend = "";
      if (datePattern.test(item)) {
        if (/^\d+\/\d+/.test(item)) {
          textToAppend = yy + "." + item;
        } else {
          textToAppend = item;
        }
      } else {
        textToAppend = eventDatePrefix + " " + item;
      }
      groupMap[7].push(textToAppend);
    }
  }

  var summaryHeaderLine = buildMemberSummaryBoardLine_(ss, name);
  var targetCols = [7, 8, 10, 11, 12];
  for (var c = 0; c < targetCols.length; c++) {
    var colNum = targetCols[c];
    var cell = sheetMember.getRange(memberRowIndex, colNum);
    var currentContent = String(cell.getValue() || "").trim();
    var lines = currentContent ? currentContent.split("\n") : [];
    
    var filteredLines = [];
    for (var l = 0; l < lines.length; l++) {
      var ln = lines[l].trim();
      if (!ln) continue;
      if (colNum === 7 && ln.indexOf("📊") === 0) continue; // 기존 요약 전광판 줄은 최신으로 교체
      if (ln.indexOf(eventDatePrefix) !== 0) {
        filteredLines.push(lines[l]);
      }
    }

    var newEventsToday = groupMap[colNum];
    for (var n = 0; n < newEventsToday.length; n++) {
      filteredLines.push(newEventsToday[n]);
    }

    // G열(7열: 이달의 학습 요약 전광판 & 최근 도서)은 최근 도서 5줄까지만 유지 + 상단에 요약 전광판 1줄 배치
    if (colNum === 7) {
      if (filteredLines.length > 5) {
        filteredLines = filteredLines.slice(filteredLines.length - 5);
      }
      if (summaryHeaderLine) {
        filteredLines.unshift(summaryHeaderLine);
      }
    }

    var newContent = filteredLines.join("\n");
    
    if (colNum === 7 || colNum === 10) {
      if (currentContent !== newContent.trim()) {
        cell.setValue(newContent);
        currentContent = newContent;
      }
      applyRichTextStyling(cell, currentContent);
    } else {
      if (currentContent !== newContent.trim()) {
        cell.setValue(newContent);
        currentContent = newContent;
      }
      var checkVal = String(cell.getValue() || "").trim();
      if (colNum === 8 && (checkVal.indexOf("코스업") !== -1 || checkVal.indexOf("레벨업") !== -1 || checkVal.indexOf("진로검사") !== -1)) {
        cell.setFontColor("#2563eb");
        cell.setFontWeight("bold");
      } else {
        cell.setFontColor(null);
        cell.setFontWeight(null);
      }
    }
  }
}

// [학습기록] 18열 시트에서 해당 학생의 이번 달 요약 1줄 생성 (G열 전광판 상단 표시용)
function buildMemberSummaryBoardLine_(ss, studentName) {
  var rs = ss.getSheetByName(RECORD_SHEET);
  if (!rs || rs.getLastRow() < 2) return "";
  var h1 = String(rs.getRange(1, 1).getValue() || "").trim();
  if (h1 !== "날짜") return "";
  var cleanTarget = String(studentName || "").replace(/\s+/g, "");
  if (!cleanTarget) return "";

  var now = new Date();
  var ymPrefix = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM");
  var mm = now.getMonth() + 1;

  var last = rs.getLastRow();
  var startRow = Math.max(2, last - 800);
  var vals = rs.getRange(startRow, 1, last - startRow + 1, 18).getValues();

  var bookSet = {};
  var diagScores = [];
  var vocabRates = [];
  var trCount = 0;

  for (var i = 0; i < vals.length; i++) {
    var v = vals[i];
    var rName = String(v[1] || "").replace(/\s+/g, "");
    if (rName !== cleanTarget) continue;
    var dStr = vocabFmtDate_(v[0]);
    if (dStr.indexOf(ymPrefix) !== 0) continue;

    var bk = String(v[5] || "").trim();
    if (bk) bookSet[bk] = true;
    var vr = vocabNum_(v[8]);
    if (vr !== "") vocabRates.push(vr);
    var ds = vocabNum_(v[9]);
    if (ds !== "") diagScores.push(ds);
    var tr = String(v[11] || "").trim();
    if (tr) trCount++;
  }

  var bookCount = Object.keys(bookSet).length;
  if (!bookCount && !vocabRates.length && !diagScores.length && !trCount) return "";

  var parts = [];
  if (bookCount || diagScores.length) {
    var avgDiag = diagScores.length ? Math.round(diagScores.reduce(function(a, b) { return a + b; }, 0) / diagScores.length) : "-";
    parts.push("도서 " + bookCount + "권" + (avgDiag !== "-" ? "(진단 " + avgDiag + "점)" : ""));
  }
  if (vocabRates.length) {
    var avgVoc = Math.round(vocabRates.reduce(function(a, b) { return a + b; }, 0) / vocabRates.length);
    parts.push("어휘 " + avgVoc + "%");
  }
  if (trCount) {
    parts.push("트레이닝 " + trCount + "회");
  }
  return "📊 [" + mm + "월 요약] " + parts.join(" · ");
}

// Google Sheets mixed text color styling function (completed books styled blue and bold line-by-line)
function applyRichTextStyling(cell, text) {
  if (!text) {
    cell.setValue("");
    cell.setFontColor(null);
    cell.setFontWeight(null);
    return;
  }
  
  // Normalize newlines to prevent indexing mismatch
  text = String(text).replace(/\r\n/g, "\n").replace(/\r/g, "\n");
  
  // Clear cell-level formatting first so it doesn't affect the entire cell
  cell.setFontColor(null);
  cell.setFontWeight(null);

  var lines = text.split("\n");
  var builder = SpreadsheetApp.newRichTextValue().setText(text);
  
  var blueStyle = SpreadsheetApp.newTextStyle()
    .setForegroundColor("#2563eb")
    .setBold(true)
    .build();
    
  var currentIndex = 0;
  var hasStyledLine = false;
  for (var i = 0; i < lines.length; i++) {
    var line = lines[i];
    var lineLength = line.length;
    try {
      if (line.indexOf("글완") !== -1 || line.indexOf("글쓰기완") !== -1 || line.indexOf("완료") !== -1 || line.indexOf("표현활동") !== -1) {
        builder.setTextStyle(currentIndex, currentIndex + lineLength, blueStyle);
        hasStyledLine = true;
      }
    } catch (e) {
      // Fail-safe check
    }
    currentIndex += lineLength + 1; // +1 for the newline
  }
  
  try {
    if (hasStyledLine) {
      cell.setRichTextValue(builder.build());
    } else {
      cell.setValue(text);
    }
  } catch (e) {
    cell.setValue(text);
  }
}

function isSameDateGAS(d1, d2) {
  var clean1 = String(d1).trim().replace(/\s+/g, "");
  var clean2 = String(d2).trim().replace(/\s+/g, "");
  if (clean1 === clean2) return true;
  var m1 = clean1.match(/(\d+)[^\d]+(\d+)/);
  var m2 = clean2.match(/(\d+)[^\d]+(\d+)/);
  if (m1 && m2) {
    return parseInt(m1[1], 10) === parseInt(m2[1], 10) && 
           parseInt(m1[2], 10) === parseInt(m2[2], 10);
  }
  return false;
}

        if (!statusRange.getValue()) {
          statusRange.setValue("보강대기");
        }
      } else {
        statusRange.setValue("");
      }
    } else if (col === 12 && row > 1) { // L열 (결석일자)
      var val = String(range.getValue()).trim();
      var dates = val ? val.split(",").map(function(d) { return d.trim(); }).filter(Boolean) : [];
      var count = dates.length;
      
      var classesStr = String(sheet.getRange(row, 4).getValue() || "");
      var grade = String(sheet.getRange(row, 2).getValue() || "");
      var baseDuration = 60;
      var match = classesStr.match(/(\d+)\s*분/);
      if (match) {
        baseDuration = parseInt(match[1], 10);
      } else {
        if (grade.indexOf("중") !== -1 || grade.indexOf("고") !== -1 || grade.indexOf("예비중") !== -1) {
          baseDuration = 90;
        } else {
          baseDuration = 60;
        }
      }
      
      var newAbsentMinsAcc = count * baseDuration;
      var currentMakeupMinsDone = parseInt(sheet.getRange(row, 34).getValue() || "0", 10) || 0; // Col AH (34)
      var newMakeupMinsRemaining = Math.max(0, newAbsentMinsAcc - currentMakeupMinsDone);

      sheet.getRange(row, 35).setValue(newMakeupMinsRemaining); // Col AI (35)
    }
  }
  
  if (sheetName === "출결보강관리" || sheetName === "오늘 출석부") {
    var startCol = range.getColumn();
    var endCol = range.getLastColumn();
    var startRow = range.getRow();
    var endRow = range.getLastRow();

    if (startCol <= 7 && endCol >= 5) {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheetStudents = ss.getSheetByName("전체시간표") || ss.getSheetByName("전체 시간표");
      for (var r = startRow; r <= endRow; r++) {
        if (r > 1) {
          var statusCell = sheet.getRange(r, 5);
          var timeCell = sheet.getRange(r, 6);
          var statusVal = String(statusCell.getValue()).trim();
          var studentName = String(sheet.getRange(r, 3).getValue()).trim();
          
          if ((statusVal === "수업중" || statusVal === "수업 중" || statusVal === "보강 수업중") && !timeCell.getValue()) {
            var now = new Date();
            var timeStr = Utilities.formatDate(now, "Asia/Seoul", "HH:mm");
            timeCell.setValue(timeStr);
          } else if (statusVal === "대기" || statusVal === "결석" || statusVal === "휴강" || !statusVal) {
            timeCell.setValue("");
          }

          // 결석일자 전체시간표 연동 로직
          if (studentName && sheetStudents) {
            var lastRowSt = sheetStudents.getLastRow();
            if (lastRowSt >= 2) {
              var studentNames = sheetStudents.getRange(2, 3, lastRowSt - 1, 1).getValues();
              var studentRowIndex = -1;
              for (var sIdx = 0; sIdx < studentNames.length; sIdx++) {
                if (String(studentNames[sIdx][0]).trim() === studentName) {
                  studentRowIndex = 2 + sIdx;
                  break;
                }
              }
              if (studentRowIndex !== -1) {
                var dateStr = String(sheet.getRange(r, 1).getValue()).trim(); // Col A
                var match = dateStr.match(/(\d+)\/(\d+)(월|화|수|목|금|토|일)/);
                var formattedAbsence = "";
                if (match) {
                  formattedAbsence = match[1] + "/" + match[2] + "(" + match[3] + ")";
                } else {
                  var today = new Date();
                  var weekdays = ["일", "월", "화", "수", "목", "금", "토"];
                  formattedAbsence = (today.getMonth() + 1) + "/" + today.getDate() + "(" + weekdays[today.getDay()] + ")";
                }

                var absentCell = sheetStudents.getRange(studentRowIndex, 12); // Col L (12th column)
                var currentAbsences = String(absentCell.getValue() || "").trim();
                var datesList = currentAbsences ? currentAbsences.split(",").map(function(d) { return d.trim(); }).filter(Boolean) : [];

                if (statusVal === "결석") {
                  var alreadyExists = false;
                  for (var i = 0; i < datesList.length; i++) {
                    if (datesList[i].replace(/\s+/g, "").indexOf(formattedAbsence.replace(/\s+/g, "")) !== -1) {
                      alreadyExists = true;
                      break;
                    }
                  }
                  if (!alreadyExists) {
                    datesList.push(formattedAbsence);
                    var newAbsencesVal = datesList.join(", ");
                    absentCell.setValue(newAbsencesVal);
                    formatAbsentCell(absentCell, newAbsencesVal);
                  }
                  
                  // ALWAYS recalculate absentMinsAcc and makeupMinsRemaining in 전체시간표 unconditionally!
                  var classesStr = String(sheetStudents.getRange(studentRowIndex, 4).getValue() || "");
                  var grade = String(sheetStudents.getRange(studentRowIndex, 2).getValue() || "");
                  var baseDuration = 60;
                  var match = classesStr.match(/(\d+)\s*분/);
                  if (match) {
                    baseDuration = parseInt(match[1], 10);
                  } else {
                    if (grade.indexOf("중") !== -1 || grade.indexOf("고") !== -1 || grade.indexOf("예비중") !== -1) {
                      baseDuration = 90;
                    } else {
                      baseDuration = 60;
                    }
                  }
                  
                  var newAbsentMinsAcc = datesList.length * baseDuration;
                  var currentMakeupMinsDone = parseInt(sheetStudents.getRange(studentRowIndex, 34).getValue() || "0", 10) || 0; // Col AH (34)
                  var newMakeupMinsRemaining = Math.max(0, newAbsentMinsAcc - currentMakeupMinsDone);
                  
                  sheetStudents.getRange(studentRowIndex, 33).setValue(newAbsentMinsAcc); // Col AG (33)
                  sheetStudents.getRange(studentRowIndex, 35).setValue(newMakeupMinsRemaining); // Col AI (35)
                } else {
                  var cleanAbsence = formattedAbsence.replace(/~/g, "").replace(/\s+/g, "");
                  var filteredList = [];
                  var modified = false;
                  for (var i = 0; i < datesList.length; i++) {
                    var cleanItem = datesList[i].replace(/~/g, "").replace(/\s+/g, "");
                    if (cleanItem.indexOf(cleanAbsence) === -1) {
                      filteredList.push(datesList[i]);
                    } else {
                      modified = true;
                    }
                  }
                  if (modified) {
                    var newAbsencesVal = filteredList.join(", ");
                    absentCell.setValue(newAbsencesVal);
                    formatAbsentCell(absentCell, newAbsencesVal);
                    
                    // Recalculate absentMinsAcc and makeupMinsRemaining in 전체시간표
                    var classesStr = String(sheetStudents.getRange(studentRowIndex, 4).getValue() || "");
                    var grade = String(sheetStudents.getRange(studentRowIndex, 2).getValue() || "");
                    var baseDuration = 60;
                    var match = classesStr.match(/(\d+)\s*분/);
                    if (match) {
                      baseDuration = parseInt(match[1], 10);
                    } else {
                      if (grade.indexOf("중") !== -1 || grade.indexOf("고") !== -1 || grade.indexOf("예비중") !== -1) {
                        baseDuration = 90;
                      } else {
                        baseDuration = 60;
                      }
                    }
                    
                    var newAbsentMinsAcc = filteredList.length * baseDuration;
                    var currentMakeupMinsDone = parseInt(sheetStudents.getRange(studentRowIndex, 34).getValue() || "0", 10) || 0; // Col AH (34)
                    var newMakeupMinsRemaining = Math.max(0, newAbsentMinsAcc - currentMakeupMinsDone);
                    
                    sheetStudents.getRange(studentRowIndex, 33).setValue(newAbsentMinsAcc); // Col AG (33)
                    sheetStudents.getRange(studentRowIndex, 35).setValue(newMakeupMinsRemaining); // Col AI (35)
                  }
                }
              }
            }
          }
          
          // 보강 은행 직접 변경 연동 로직
          try {
            processMakeupBankOnSheetEdit(sheet, r);
          } catch(err) {
            console.error("Direct sheet edit deduction failed for row " + r + ": " + err);
          }
        }
      }
    }
    
    // 2. 숫자, 이벤트 등 입력 시 회원 분석 시트로 자동 누적
    if (startCol <= 11 && endCol >= 8) {
      for (var r = startRow; r <= endRow; r++) {
        if (r > 1) {
          accumulateDailyEventToMemberAnalysis(sheet, r);
        }
      }
    }
  }
}

// 구글 시트에서 직접 등원여부를 '수업완료'/'보강완료'로 변경하거나 보강사유를 넣었을 때 실시간 보강 은행 정산
function processMakeupBankOnSheetEdit(sheet, r) {
  var statusCell = sheet.getRange(r, 5);
  var reasonCell = sheet.getRange(r, 7);
  var statusVal = String(statusCell.getValue()).trim();
  var reasonVal = String(reasonCell.getValue()).trim();
  var studentName = String(sheet.getRange(r, 3).getValue()).trim();
  
  if (!studentName) return;
  if (statusVal !== "수업완료" && statusVal !== "보강완료") return;
  if (!reasonVal) return;
  if (reasonVal.indexOf("[차감완료]") !== -1) return;
  
  var isMakeup = (reasonVal === "보강 수업");
  var isExtension = (reasonVal.indexOf("연장 (") === 0);
  
  if (!isMakeup && !isExtension) return;
  
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetStudents = ss.getSheetByName("전체시간표") || ss.getSheetByName("전체 시간표");
  if (!sheetStudents) return;
  
  var lastRowSt = sheetStudents.getLastRow();
  if (lastRowSt < 2) return;
  
  var studentData = sheetStudents.getRange(2, 1, lastRowSt - 1, 36).getValues();
  var studentRowIndex = -1;
  var studentRowData = null;
  for (var sIdx = 0; sIdx < studentData.length; sIdx++) {
    if (String(studentData[sIdx][2]).trim() === studentName) {
      studentRowIndex = 2 + sIdx;
      studentRowData = studentData[sIdx];
      break;
    }
  }
  
  if (studentRowIndex === -1 || !studentRowData) return;
  
  var classesStr = String(studentRowData[3] || "");
  var grade = String(studentRowData[1] || "");
  var baseDuration = 60;
  var match = classesStr.match(/(60|70|90|100|120|180)\s*분/);
  if (match) {
    baseDuration = parseInt(match[1], 10);
  } else {
    if (grade.indexOf("중") !== -1 || grade.indexOf("고") !== -1 || grade.indexOf("예비중") !== -1) {
      baseDuration = 90;
    } else {
      baseDuration = 60;
    }
  }
  
  var currentMakeupMinsDone = parseInt(studentRowData[33], 10) || 0;
  var currentAbsentMinsAcc = parseInt(studentRowData[32], 10) || 0;
  var todayExtensionMins = parseInt(studentRowData[35], 10) || 0;
  
  var addedMins = 0;
  if (isMakeup) {
    addedMins = baseDuration + todayExtensionMins;
  } else if (isExtension) {
    var matchExt = reasonVal.match(/\d+/);
    addedMins = matchExt ? parseInt(matchExt[0], 10) : 0;
  }
  
  if (addedMins <= 0) return;
  
  var newMakeupMinsDone = currentMakeupMinsDone + addedMins;
  
  var absentDatesStr = String(studentRowData[11] || "");
  var datesList = absentDatesStr ? absentDatesStr.split(",").map(function(d) { return d.trim(); }).filter(Boolean) : [];
  
  var targetStrikethroughsCount = Math.floor(newMakeupMinsDone / baseDuration);
  var updatedDates = [];
  var strikedCount = 0;
  for (var i = 0; i < datesList.length; i++) {
    var item = datesList[i];
    var isStriked = (item.indexOf("~~") === 0 && item.slice(-2) === "~~");
    var cleanDate = isStriked ? item.slice(2, -2) : item;
    if (strikedCount < targetStrikethroughsCount) {
      updatedDates.push("~~" + cleanDate + "~~");
      strikedCount++;
    } else {
      updatedDates.push(cleanDate);
    }
  }
  var newAbsentDatesVal = updatedDates.join(", ");
  
  var newMakeupMinsRemaining = Math.max(0, currentAbsentMinsAcc - newMakeupMinsDone);
  
  sheetStudents.getRange(studentRowIndex, 34).setValue(newMakeupMinsDone);
  sheetStudents.getRange(studentRowIndex, 35).setValue(newMakeupMinsRemaining);
  sheetStudents.getRange(studentRowIndex, 12).setValue(newAbsentDatesVal);
  
  if (isMakeup) {
    sheetStudents.getRange(studentRowIndex, 32).setValue("보강완료");
  }
  if (isExtension || todayExtensionMins > 0) {
    sheetStudents.getRange(studentRowIndex, 36).setValue(0);
  }
  
  reasonCell.setValue(reasonVal + " [차감완료]");
}

// 오답 개수 문자열 결합 버그('10/2' -> 102, '10080') 방지 파서
function vocabParseErrorNum_(v) {
  if (v === "" || v === null || v === undefined) return "";
  if (typeof v === "number") return (v >= 0 && v <= 100) ? v : 0;
  var s = String(v).trim();
  if (!s) return "";
  var slashMatch = s.match(/^\s*(\d+)\s*\/\s*(\d+)/);
  if (slashMatch) {
    var n1 = parseInt(slashMatch[1], 10);
    var n2 = parseInt(slashMatch[2], 10);
    if (n2 <= n1 && n2 < 50) return n2;
    return n1 < 50 ? n1 : 0;
  }
  var jungMatch = s.match(/중\s*(\d+)/);
  if (jungMatch) return parseInt(jungMatch[1], 10) || 0;
  var numMatch = s.match(/(\d+)/);
  if (!numMatch) return "";
  var val = parseInt(numMatch[1], 10);
  if (isNaN(val)) return "";
  // 기존에 잘못 들어간 '10080', '101' 같은 비정상 오답수 자동 보정
  if (val > 50) {
    if (s.length === 3 && s.charAt(0) === "1" && s.charAt(1) === "0") return parseInt(s.charAt(2), 10);
    return 0;
  }
  return val;
}

function vocabParseDate_(s) {
  if (s instanceof Date) return new Date(s.getFullYear(), s.getMonth(), s.getDate());
  var str = String(s || "").trim();
  if (/^\d{4}-\d{2}-\d{2}/.test(str)) {
    return Utilities.parseDate(str.substring(0, 10), Session.getScriptTimeZone(), "yyyy-MM-dd");
  }
  var now = new Date();
  return new Date(now.getFullYear(), now.getMonth(), now.getDate());
}

function vocabFmtDate_(d) {
  if (!(d instanceof Date)) d = vocabParseDate_(d);
  return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd");
}

function vocabReadRows_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(VOCAB_SHEET);
  if (!sheet || sheet.getLastRow() < 2) return [];
  var values = sheet.getRange(2, 1, sheet.getLastRow() - 1, 10).getValues();
  var rows = [];
  for (var i = 0; i < values.length; i++) {
    var v = values[i];
    var use = String(v[9] === "" ? "O" : v[9]).trim().toUpperCase();
    if (use === "X") continue;                       // 사용 = X 는 문제에서 제외
    var word = String(v[4] || "").trim();
    var meaning = String(v[6] || "").trim();
    if (!word || !meaning) continue;
    rows.push({ level: String(v[1] || "").trim(), title: String(v[2] || "").trim(), w: word, f: String(v[5] || "").trim() || word, m: meaning, e: String(v[7] || "").trim() });
  }
  return rows;
}

function vocabGroupBooks_(rows) {
  var order = [], map = {};
  for (var i = 0; i < rows.length; i++) {
    var r = rows[i];
    var key = r.level + "|" + r.title;
    if (!map[key]) { map[key] = { level: r.level, title: r.title, words: [] }; order.push(key); }
    map[key].words.push({ w: r.w, f: r.f, m: r.m, e: r.e });
  }
  var books = [];
  for (var k = 0; k < order.length; k++) books.push(map[order[k]]);
  return books;
}

// CacheService 는 값 하나당 100KB 제한이라 조각내어 저장
function vocabCachePut_(key, obj) {
  try {
    var json = JSON.stringify(obj);
    var size = 20000, parts = {}, n = 0;
    for (var p = 0; p < json.length; p += size) { parts[key + "#" + n] = json.substring(p, p + size); n++; }
    parts[key + "#n"] = String(n);
    CacheService.getScriptCache().putAll(parts, 21600);
  } catch (err) { /* 캐시 실패는 무시 (시트를 직접 읽으면 됨) */ }
}

function vocabCacheGet_(key) {
  try {
    var cache = CacheService.getScriptCache();
    var n = cache.get(key + "#n");
    if (!n) return null;
    var names = [];
    for (var i = 0; i < Number(n); i++) names.push(key + "#" + i);
    var got = cache.getAll(names);
    var json = "";
    for (var j = 0; j < names.length; j++) {
      if (got[names[j]] === undefined) return null;
      json += got[names[j]];
    }
    return JSON.parse(json);
  } catch (err) { return null; }
}

function vocabGetIndex_() {
  var key = "vocabIdx_" + vocabVersion_();
  var cached = vocabCacheGet_(key);
  if (cached) return cached;
  var books = vocabGroupBooks_(vocabReadRows_());
  var items = [];
  for (var i = 0; i < books.length; i++) items.push({ t: books[i].title, l: books[i].level, n: books[i].words.length });
  var out = { type: "vocabIndex", version: vocabVersion_(), items: items };
  vocabCachePut_(key, out);
  return out;
}

function vocabGetLevel_(level) {
  if (VOCAB_LEVELS.indexOf(level) === -1) return { type: "vocabError", error: "unknown level: " + level };
  var key = "vocabLv_" + vocabVersion_() + "_" + level;
  var cached = vocabCacheGet_(key);
  if (cached) return cached;
  var rows = vocabReadRows_().filter(function (r) { return r.level === level; });
  var out = { type: "vocabLevel", version: vocabVersion_(), level: level, books: vocabGroupBooks_(rows) };
  vocabCachePut_(key, out);
  return out;
}

// doGet(e) 맨 앞에서 호출: 어휘 요청이면 응답을 돌려주고, 아니면 null(기존 동작 그대로)
function handleVocabGet(e) {
  var action = e && e.parameter && e.parameter.action;
  if (action !== "getVocab" && action !== "getVocabIndex") return null;
  var out;
  try {
    out = (action === "getVocabIndex") ? vocabGetIndex_() : vocabGetLevel_(String(e.parameter.level || ""));
  } catch (err) {
    out = { type: "vocabError", error: String(err) };
  }
  return vocabJson_(out);
}

// 기존 17열(A:날짜, B:시간, C:이름, D:종류, ... 또는 A:시간, B:날짜, C:이름...) 시트를 18열(방식 1 가로 병합)로 안전 변환
function vocabIsTimeLikeValue_(val) {
  if (!val) return false;
  if (val instanceof Date) return true;
  var s = String(val).trim();
  if (s.indexOf("표준시") !== -1 || s.indexOf("GMT") !== -1) return true;
  if (/^\d{1,2}:\d{2}/.test(s)) return true;
  if (/^(오전|오후)\s*\d{1,2}:\d{2}/.test(s)) return true;
  return false;
}

function vocabMigrate17To18Cols_(ss, sheet) {
  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow < 1) return false;
  var headers = sheet.getRange(1, 1, 1, Math.min(Math.max(lastCol, 4), 18)).getValues()[0];
  var h1 = String(headers[0] || "").trim();
  var h2 = String(headers[1] || "").trim();
  var h3 = String(headers[2] || "").trim();
  var h4 = String(headers[3] || "").trim();

  var oldValues = lastRow >= 2 ? sheet.getRange(2, 1, lastRow - 1, Math.min(Math.max(lastCol, 17), 18)).getValues() : [];

  // 데이터 행에 B열이 '시간(Date/표준시)'이고 C열이 '학생이름'인 구버전 행이 섞여 있는지 정밀 검사
  var hasLegacyTimeInColB = false;
  for (var chk = 0; chk < oldValues.length; chk++) {
    if (vocabIsTimeLikeValue_(oldValues[chk][1]) && String(oldValues[chk][2] || "").trim()) {
      hasLegacyTimeInColB = true;
      break;
    }
  }

  var isOldHeader = (h1 === "시간" || h2 === "시간" || h4 === "종류" || (h1 === "날짜" && h3 === "이름"));
  if (!isOldHeader && !hasLegacyTimeInColB) {
    return false;
  }

  // 안전 백업 시트 생성
  var backupName = RECORD_SHEET + "_17열백업";
  if (!ss.getSheetByName(backupName)) {
    var copied = sheet.copyTo(ss);
    copied.setName(backupName);
  }

  // 날짜+이름 그룹별로 묶어서 방식 1(가로 병합 + 초과 시 2·3번째 줄)로 재구성
  var groups = {};
  var groupOrder = [];
  var already18Rows = [];

  for (var i = 0; i < oldValues.length; i++) {
    var v = oldValues[i];
    var isLegacyRow = isOldHeader || vocabIsTimeLikeValue_(v[1]);

    if (!isLegacyRow) {
      // 이미 18열 포맷으로 들어온 정상 행(B열이 학생이름)은 그대로 보존
      var nm18 = String(v[1] || "").replace(/\s+/g, "");
      if (nm18 && !vocabIsTimeLikeValue_(v[1])) {
        var row18 = v.slice(0, 18);
        while (row18.length < 18) row18.push("");
        row18[0] = vocabParseDate_(v[0]);
        row18[1] = nm18;
        already18Rows.push(row18);
      }
      continue;
    }

    var type = String(v[3] || "").trim();
    if (type === "어휘오답") continue; // 불필요한 어휘오답 행 제거!

    // A열(v[0])과 B열(v[1]) 중 실제 날짜(2020년 이후)를 우선 선택
    var rawDate = v[0];
    if (v[0] instanceof Date && v[0].getFullYear() < 2000 && v[1] instanceof Date && v[1].getFullYear() >= 2000) {
      rawDate = v[1];
    } else if (!v[0] && v[1]) {
      rawDate = v[1];
    }
    var dateObj = vocabParseDate_(rawDate);
    var dateKey = vocabFmtDate_(dateObj);
    var name = String(v[2] || "").replace(/\s+/g, "");
    if (!name || vocabIsTimeLikeValue_(v[2])) continue;

    var gKey = dateKey + "||" + name;
    if (!groups[gKey]) {
      groups[gKey] = { dateObj: dateObj, name: name, records: [] };
      groupOrder.push(gKey);
    }
    var rec = {
      date: dateKey,
      type: type,
      level: String(v[4] || "").trim(),
      book: String(v[5] || "").trim(),
      total: v[6],
      firstCorrect: v[7],
      rate: v[8],
      score: v[9],
      training: String(v[10] || "").trim(),
      detail: String(v[11] || "").trim(),
      errors: vocabParseErrorNum_(v[12]),
      understanding: String(v[13] || "").trim(),
      wrongWords: String(v[14] || "").trim(),
      step2Memo: String(v[15] || "").trim(),
      literacyMemo: String(v[15] || "").trim(),
      reflection: String(v[16] || "").trim()
    };
    groups[gKey].records.push(rec);
  }

  sheet.clear();
  vocabStyleRecordSheet_(sheet);

  if (already18Rows.length > 0) {
    sheet.getRange(2, 1, already18Rows.length, 18).setValues(already18Rows);
  }

  for (var g = 0; g < groupOrder.length; g++) {
    var grp = groups[groupOrder[g]];
    vocabUpsertStudentDayRecords_(sheet, grp.name, grp.records);
  }
  return true;
}

// 출결보강관리(오늘 출석부)·학생수업관리·회원분석의 수업내용(K열/F열) 문자열에서 숙제현황 및 발자국 4종 추출
function parseDailyContentsFootprints_(contentsStr) {
  var s = String(contentsStr || "").trim();
  if (!s) return null;
  var hwStatus = "";
  if (s.indexOf("숙제완료") !== -1) hwStatus = "숙제완료";
  else if (s.indexOf("숙제미완") !== -1) hwStatus = "숙제미완";
  else if (s.indexOf("미지참") !== -1) hwStatus = "미지참";
  else if (s.indexOf("학원에서 풀기") !== -1 || s.indexOf("학원에서풀기") !== -1) hwStatus = "학원에서풀기";

  var hwBookMatch = s.match(/(?:숙제완료|숙제미완|미지참|학원에서\s*풀기)\s*\(([^)]+)\)/);
  var hwBook = hwBookMatch ? String(hwBookMatch[1] || "").trim() : "";

  var step1Match = s.match(/\[성실발자국:\s*([^\]]+)\]/);
  var rawStep1 = step1Match ? String(step1Match[1] || "").trim() : "";
  var step1Memo = hwBook ? (rawStep1 ? "[" + hwBook + "] " + rawStep1 : "[" + hwBook + "]") : rawStep1;

  var step2Match = s.match(/\[독서발자국:\s*([^\]]+)\]/);
  var step2Memo = step2Match ? String(step2Match[1] || "").trim() : "";

  var litParts = [];
  var step3Match = s.match(/\[메모:\s*([^\]]+)\]/);
  if (step3Match && String(step3Match[1] || "").trim()) litParts.push(String(step3Match[1]).trim());
  var step4Match = s.match(/\[문해력발자국:\s*([^\]]+)\]/);
  if (step4Match && String(step4Match[1] || "").trim()) litParts.push(String(step4Match[1]).trim());
  var literacyMemo = litParts.join(" / ");

  var reflMatch = s.match(/\[소감\]\s*([^|\r\n]+)/);
  var reflection = reflMatch ? String(reflMatch[1] || "").trim() : "";

  if (!hwStatus && !step1Memo && !step2Memo && !literacyMemo && !reflection) return null;
  return {
    hwStatus: hwStatus,
    step1Memo: step1Memo,
    step2Memo: step2Memo,
    literacyMemo: literacyMemo,
    reflection: reflection
  };
}

// 출결보강관리(오늘 출석부)·학생수업관리·회원분석에 저장된 발자국 기록을 [학습기록] 18열 시트의 빈 발자국 칸에 자동 복원/동기화
function syncFootprintsFromDailySheetsToLearningRecords_(ss, sheetRec) {
  try {
    if (!sheetRec || sheetRec.getLastRow() < 2) return;
    var recLastRow = sheetRec.getLastRow();
    var recData = sheetRec.getRange(2, 1, recLastRow - 1, 18).getValues();
    var dirty = false;

    function applyFootprintToStudentRows(studentName, dateStrHint, fp) {
      if (!studentName || !fp) return;
      var cleanSt = String(studentName).replace(/\s+/g, "");
      if (!cleanSt) return;

      // 1순위: 날짜+이름 일치 행 찾기, 없으면 해당 학생의 가장 최근 행 찾기
      var targetIdx = -1;
      if (dateStrHint) {
        for (var i = recData.length - 1; i >= 0; i--) {
          var rName = String(recData[i][1] || "").replace(/\s+/g, "");
          var rDate = vocabFmtDate_(recData[i][0]);
          if (rName === cleanSt && rDate === dateStrHint) {
            targetIdx = i; // 첫 줄 우선(역순 탐색이므로 계속 갱신하여 해당 날짜 첫 줄 선택)
          }
        }
      }
      if (targetIdx === -1) {
        for (var j = recData.length - 1; j >= 0; j--) {
          var rName2 = String(recData[j][1] || "").replace(/\s+/g, "");
          if (rName2 === cleanSt) {
            var latestDate = vocabFmtDate_(recData[j][0]);
            for (var k = 0; k < recData.length; k++) {
              if (String(recData[k][1] || "").replace(/\s+/g, "") === cleanSt && vocabFmtDate_(recData[k][0]) === latestDate) {
                targetIdx = k;
                break;
              }
            }
            break;
          }
        }
      }
      if (targetIdx === -1) return;

      var row = recData[targetIdx];
      if (fp.hwStatus && !String(row[2] || "").trim()) { row[2] = fp.hwStatus; dirty = true; }
      if (fp.step1Memo && !String(row[3] || "").trim()) { row[3] = fp.step1Memo; dirty = true; }
      if (fp.step2Memo && !String(row[10] || "").trim()) { row[10] = fp.step2Memo; dirty = true; }
      if (fp.literacyMemo && !String(row[15] || "").trim()) { row[15] = fp.literacyMemo; dirty = true; }
      if (fp.reflection && !String(row[16] || "").trim()) { row[16] = fp.reflection; dirty = true; }
    }

    var curYear = new Date().getFullYear();
    function parseSlashDateHint(rawD) {
      if (rawD instanceof Date && !isNaN(rawD.getTime())) return vocabFmtDate_(rawD);
      var s = String(rawD || "").trim();
      var m = s.match(/(?:(\d{2,4})[\.\-\/])?(\d{1,2})[\.\-\/](\d{1,2})/);
      if (m) {
        var y = m[1] ? (m[1].length === 2 ? 2000 + parseInt(m[1], 10) : parseInt(m[1], 10)) : curYear;
        var mo = parseInt(m[2], 10);
        var da = parseInt(m[3], 10);
        return y + "-" + (mo < 10 ? "0" + mo : mo) + "-" + (da < 10 ? "0" + da : da);
      }
      return "";
    }

    // 1) 출결보강관리 / 오늘 출석부 스캔
    var dailyNames = ["출결보강관리", "오늘 출석부", "오늘출석부"];
    for (var d = 0; d < dailyNames.length; d++) {
      var shD = ss.getSheetByName(dailyNames[d]);
      if (shD && shD.getLastRow() >= 3) {
        var dRows = shD.getRange(3, 1, shD.getLastRow() - 2, Math.min(shD.getLastColumn(), 11)).getValues();
        for (var r1 = 0; r1 < dRows.length; r1++) {
          var nm1 = String(dRows[r1][2] || "").replace(/\s+/g, "");
          var cont1 = String(dRows[r1][10] || "").trim();
          var fp1 = parseDailyContentsFootprints_(cont1);
          if (nm1 && fp1) {
            applyFootprintToStudentRows(nm1, parseSlashDateHint(dRows[r1][0]), fp1);
          }
        }
      }
    }

    // 2) 학생수업관리 / 누적일일수업관리 최근 300행 스캔
    var accumNames = ["학생수업관리", "누적일일수업관리"];
    for (var a = 0; a < accumNames.length; a++) {
      var shA = ss.getSheetByName(accumNames[a]);
      if (shA && shA.getLastRow() >= 2) {
        var stRow = Math.max(2, shA.getLastRow() - 300);
        var aRows = shA.getRange(stRow, 1, shA.getLastRow() - stRow + 1, Math.min(shA.getLastColumn(), 11)).getValues();
        for (var r2 = 0; r2 < aRows.length; r2++) {
          var nm2 = String(aRows[r2][2] || "").replace(/\s+/g, "");
          var cont2 = String(aRows[r2][10] || "").trim();
          var fp2 = parseDailyContentsFootprints_(cont2);
          if (nm2 && fp2) {
            applyFootprintToStudentRows(nm2, parseSlashDateHint(aRows[r2][0]), fp2);
          }
        }
      }
    }

    // 3) 회원 분석 / 회원 분석_백업 F열 스캔
    var memNames = ["회원 분석_백업", "회원 분석", "회원분석", "월별상담내역"];
    for (var mIdx = 0; mIdx < memNames.length; mIdx++) {
      var shM = ss.getSheetByName(memNames[mIdx]);
      if (shM && shM.getLastRow() >= 2) {
        var mRows = shM.getRange(2, 1, shM.getLastRow() - 1, Math.min(shM.getLastColumn(), 7)).getValues();
        for (var r3 = 0; r3 < mRows.length; r3++) {
          var nm3 = String(mRows[r3][1] || "").replace(/\s+/g, "");
          var fText = String(mRows[r3][5] || "").trim();
          if (!nm3 || !fText) continue;
          var fLines = fText.split(/\r?\n/);
          for (var l = 0; l < fLines.length; l++) {
            var fp3 = parseDailyContentsFootprints_(fLines[l]);
            if (fp3) {
              applyFootprintToStudentRows(nm3, parseSlashDateHint(fLines[l]), fp3);
            }
          }
        }
      }
    }

    if (dirty) {
      sheetRec.getRange(2, 1, recData.length, 18).setValues(recData);
    }
  } catch (e) {}
}

function vocabEnsureRecordSheet_(ss) {
  var sheet = ss.getSheetByName(RECORD_SHEET);
  if (!sheet) {
    sheet = ss.insertSheet(RECORD_SHEET);
    vocabStyleRecordSheet_(sheet);
  } else {
    var migrated = vocabMigrate17To18Cols_(ss, sheet);
    if (!migrated) {
      var h2 = String(sheet.getRange(1, 2).getValue() || "").trim();
      var h3 = String(sheet.getRange(1, 3).getValue() || "").trim();
      if (h2 !== "이름" || h3 !== "숙제현황") {
        vocabStyleRecordSheet_(sheet);
      }
    }
    syncFootprintsFromDailySheetsToLearningRecords_(ss, sheet);
  }
  return sheet;
}

function vocabMakeEmptyRow18_(dateObj, name) {
  return [
    dateObj, name, "", "", "", "", "", "", "", "", "", "", "", "", "", "", "", ""
  ];
}

// 방식 1: (날짜, 이름) 기준 첫 줄 가로 병합 + 같은 날 도서/어휘나 트레이닝이 2~3개면 두 줄·세 줄 자동 확장
function vocabUpsertStudentDayRecords_(sheet, name, records) {
  if (!name || !records || !records.length) return { added: 0, updated: 0 };

  // 날짜별로 입력 레코드 묶기
  var byDate = {};
  var dateKeys = [];
  for (var i = 0; i < records.length; i++) {
    var r = records[i] || {};
    var rt = String(r.type || "").trim();
    if (rt === "어휘오답") continue;
    var dObj = vocabParseDate_(r.date);
    var dKey = vocabFmtDate_(dObj);
    if (!byDate[dKey]) {
      byDate[dKey] = { dateObj: dObj, items: [] };
      dateKeys.push(dKey);
    }
    byDate[dKey].items.push(r);
  }

  var lastRow = sheet.getLastRow();
  var allData = lastRow >= 2 ? sheet.getRange(2, 1, lastRow - 1, 18).getValues() : [];
  var addedCount = 0;
  var updatedCount = 0;

  for (var dIdx = 0; dIdx < dateKeys.length; dIdx++) {
    var dKey = dateKeys[dIdx];
    var dObj = byDate[dKey].dateObj;
    var items = byDate[dKey].items;

    // 시트에서 해당 (날짜, 이름)의 기존 행 인덱스(0-based in allData) 찾기
    var matchIndices = [];
    for (var rowIdx = 0; rowIdx < allData.length; rowIdx++) {
      var rDate = allData[rowIdx][0] ? vocabFmtDate_(allData[rowIdx][0]) : "";
      var rName = String(allData[rowIdx][1] || "").replace(/\s+/g, "");
      if (rDate === dKey && rName === name) {
        matchIndices.push(rowIdx);
      }
    }

    // 자정(00시~08시) 이후 테스트 또는 일일요약(발자국)만 전송된 경우: 해당 학생의 직전 수업일 행(도서명 일치 또는 최근 행)에 자동 병합
    if (matchIndices.length === 0 && allData.length > 0) {
      var incomingBooks = {};
      var hasOnlySummaryOrMatchingBook = true;
      for (var ib = 0; ib < items.length; ib++) {
        var bNorm = vocabNormBook_(items[ib].book);
        if (bNorm) incomingBooks[bNorm] = true;
      }
      var latestStudentDate = "";
      for (var fb = allData.length - 1; fb >= 0; fb--) {
        var fbName = String(allData[fb][1] || "").replace(/\s+/g, "");
        if (fbName !== name) continue;
        var fbDate = allData[fb][0] ? vocabFmtDate_(allData[fb][0]) : "";
        var fbBook = vocabNormBook_(allData[fb][5]);
        if (fbBook && incomingBooks[fbBook]) {
          latestStudentDate = fbDate;
          break;
        }
        if (!latestStudentDate && (items.length === 1 && items[0].type === "일일요약" || (new Date()).getHours() < 8)) {
          latestStudentDate = fbDate;
        }
      }
      if (latestStudentDate) {
        for (var rowIdx2 = 0; rowIdx2 < allData.length; rowIdx2++) {
          var rDate2 = allData[rowIdx2][0] ? vocabFmtDate_(allData[rowIdx2][0]) : "";
          var rName2 = String(allData[rowIdx2][1] || "").replace(/\s+/g, "");
          if (rDate2 === latestStudentDate && rName2 === name) {
            matchIndices.push(rowIdx2);
            dObj = vocabParseDate_(latestStudentDate);
          }
        }
      }
    }

    // 작업용 행 배열(studentRows): 기존 행이 있으면 가져오고, 없으면 빈 첫 줄 1개 생성
    var studentRows = [];
    for (var m = 0; m < matchIndices.length; m++) {
      var clone = allData[matchIndices[m]].slice(0, 18);
      while (clone.length < 18) clone.push("");
      clone[0] = dObj;
      clone[1] = name;
      studentRows.push(clone);
    }
    if (!studentRows.length) {
      studentRows.push(vocabMakeEmptyRow18_(dObj, name));
    }

    function ensureRow(idx) {
      while (studentRows.length <= idx) {
        studentRows.push(vocabMakeEmptyRow18_(dObj, name));
      }
      return studentRows[idx];
    }

    for (var k = 0; k < items.length; k++) {
      var it = items[k];
      var type = String(it.type || "").trim();

      // 1) 일일요약 (숙제현황 C열, 성실발자국 D열, 독서발자국 K열, 문해력발자국 P열, 나의기록 Q열) -> 첫 줄(Row 0)에 기록
      if (type === "일일요약") {
        var r0 = ensureRow(0);
        if (it.hwStatus) r0[2] = String(it.hwStatus).trim();
        if (it.step1Memo) r0[3] = String(it.step1Memo).trim();
        if (it.step2Memo && !r0[10]) r0[10] = String(it.step2Memo).trim();
        if (it.literacyMemo && !r0[15]) r0[15] = String(it.literacyMemo).trim();
        if (it.reflection) r0[16] = String(it.reflection).trim();
        continue;
      }

      // 2) 어휘퀴즈 (E:레벨구분, F:도서명/어휘학습, G:어휘문제, H:어휘정답, I:어휘정답률, R:틀린어휘)
      if (type === "어휘퀴즈") {
        var bookName = String(it.book || "").trim();
        var normBook = vocabNormBook_(bookName);
        var targetIdx = -1;
        for (var s = 0; s < studentRows.length; s++) {
          var rowBook = vocabNormBook_(studentRows[s][5]);
          // 같은 책 이름이 이미 있거나(진단점수가 먼저 찍힌 경우 등), 어휘 칸(F,G)이 비어 있는 첫 행
          if ((normBook && rowBook === normBook) || (!studentRows[s][5] && studentRows[s][6] === "")) {
            targetIdx = s;
            break;
          }
        }
        if (targetIdx === -1) targetIdx = studentRows.length;
        var vRow = ensureRow(targetIdx);
        if (it.level) vRow[4] = String(it.level).trim();
        if (bookName) vRow[5] = bookName;
        if (it.total !== "" && it.total !== undefined) vRow[6] = vocabNum_(it.total);
        if (it.firstCorrect !== "" && it.firstCorrect !== undefined) vRow[7] = vocabNum_(it.firstCorrect);
        if (it.rate !== "" && it.rate !== undefined) vRow[8] = vocabNum_(it.rate);
        if (it.wrongWords !== undefined && String(it.wrongWords).trim() !== "") {
          vRow[17] = vocabCleanWrongWordsStr_(it.wrongWords);
        }
        continue;
      }

      // 3) 진단 (E:레벨구분, F:도서명/어휘학습, J:진단점수, K:독서발자국)
      // 어휘퀴즈와 책 이름이 같으면 같은 줄 J열·K열에 병합! 다른 책이면 빈 줄 또는 다음 줄에 기록
      if (type === "진단") {
        var dBook = String(it.book || "").trim();
        var normDBook = vocabNormBook_(dBook);
        var dIdxTarget = -1;
        for (var s2 = 0; s2 < studentRows.length; s2++) {
          var existBook = vocabNormBook_(studentRows[s2][5]);
          if (normDBook && existBook === normDBook) {
            dIdxTarget = s2;
            break;
          }
        }
        if (dIdxTarget === -1) {
          for (var s3 = 0; s3 < studentRows.length; s3++) {
            if (!studentRows[s3][5] && studentRows[s3][9] === "") {
              dIdxTarget = s3;
              break;
            }
          }
        }
        if (dIdxTarget === -1) dIdxTarget = studentRows.length;
        var dRow = ensureRow(dIdxTarget);
        if (it.level && !dRow[4]) dRow[4] = String(it.level).trim();
        if (dBook && !dRow[5]) dRow[5] = dBook;
        if (it.score !== "" && it.score !== undefined) dRow[9] = vocabNum_(it.score);
        if (it.step2Memo) dRow[10] = String(it.step2Memo).trim();
        continue;
      }

      // 4) 트레이닝 (L:트레이닝, M:권수/차시, N:오답개수, O:이해도, P:문해력발자국)
      // 첫 줄 L열이 비어 있으면 첫 줄에, 이미 다른 트레이닝이 들어있으면 2번째·3번째 줄에 각각 깔끔하게 기록!
      if (type === "트레이닝") {
        var trName = String(it.training || "").trim();
        var trDetail = String(it.detail || "").trim();
        var tIdxTarget = -1;
        for (var s4 = 0; s4 < studentRows.length; s4++) {
          var existTr = String(studentRows[s4][11] || "").trim();
          var existDt = String(studentRows[s4][12] || "").trim();
          if (existTr === trName && existDt === trDetail && trName !== "") {
            tIdxTarget = s4;
            break;
          }
          if (!existTr && tIdxTarget === -1) {
            tIdxTarget = s4;
          }
        }
        if (tIdxTarget === -1) tIdxTarget = studentRows.length;
        var tRow = ensureRow(tIdxTarget);
        if (trName) tRow[11] = trName;
        if (trDetail) tRow[12] = trDetail;
        var parsedErr = vocabParseErrorNum_(it.errors);
        if (parsedErr !== "") tRow[13] = parsedErr;
        if (it.understanding) tRow[14] = String(it.understanding).trim();
        if (it.literacyMemo) tRow[15] = String(it.literacyMemo).trim();
        if (it.hwStatus && !tRow[2]) tRow[2] = String(it.hwStatus).trim();
        if (it.step1Memo && !tRow[3]) tRow[3] = String(it.step1Memo).trim();
        if (it.reflection && !tRow[16]) tRow[16] = String(it.reflection).trim();
      }
    }

    // 기존 행 업데이트 + 초과 행은 시트 하단에 추가
    for (var u = 0; u < studentRows.length; u++) {
      if (u < matchIndices.length) {
        var sheetRowNum = matchIndices[u] + 2;
        sheet.getRange(sheetRowNum, 1, 1, 18).setValues([studentRows[u]]);
        allData[matchIndices[u]] = studentRows[u];
        updatedCount++;
      } else {
        var newRowNum = sheet.getLastRow() + 1;
        sheet.getRange(newRowNum, 1, 1, 18).setValues([studentRows[u]]);
        allData.push(studentRows[u]);
        addedCount++;
      }
    }
  }

  return { added: addedCount, updated: updatedCount };
}

// 학생 1명의 [회원 분석] G열(이달의 학습 요약 전광판 & 최근 도서)을 최신 [학습기록] 기준으로 즉시 갱신
function refreshMemberAnalysisSummaryBoardForStudent_(ss, studentName) {
  try {
    var sheetMember = getMemberAnalysisSheet(ss);
    if (!sheetMember || sheetMember.getLastRow() < 2) return;
    var cleanTarget = String(studentName || "").replace(/\s+/g, "");
    if (!cleanTarget) return;
    var lastRow = sheetMember.getLastRow();
    var names = sheetMember.getRange(2, 2, lastRow - 1, 1).getValues();
    var memberRowIndex = -1;
    for (var i = 0; i < names.length; i++) {
      if (String(names[i][0] || "").replace(/\s+/g, "") === cleanTarget) {
        memberRowIndex = 2 + i;
        break;
      }
    }
    if (memberRowIndex === -1) return;
    var summaryLine = buildMemberSummaryBoardLine_(ss, cleanTarget);
    if (!summaryLine) return;

    var cell = sheetMember.getRange(memberRowIndex, 7);
    var cur = String(cell.getValue() || "").trim();
    var lines = cur ? cur.split("\n") : [];
    var bookLines = [];
    for (var l = 0; l < lines.length; l++) {
      var ln = lines[l].trim();
      if (!ln || ln.indexOf("📊") === 0) continue;
      bookLines.push(ln);
    }
    if (bookLines.length > 5) {
      bookLines = bookLines.slice(bookLines.length - 5);
    }
    bookLines.unshift(summaryLine);
    var nextVal = bookLines.join("\n");
    if (cur !== nextVal) {
      applyRichTextStyling(cell, nextVal);
    }
  } catch (e) {}
}

function handleAddLearningRecord(postData) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = vocabEnsureRecordSheet_(ss);
    var name = String(postData.name || "").replace(/\s+/g, "");
    if (!name || !postData.records || !postData.records.length) {
      return vocabJson_({ status: "error", message: "name/records required" });
    }
    var res = vocabUpsertStudentDayRecords_(sheet, name, postData.records);
    refreshMemberAnalysisSummaryBoardForStudent_(ss, name);
    return vocabJson_({ status: "success", added: res.added, updated: res.updated });
  } catch (err) {
    return vocabJson_({ status: "error", message: String(err) });
  } finally {
    lock.releaseLock();
  }
}

// ---------- 메뉴 12번: 회원 분석 시트 다이어트 (F·G열 과거 일일기록 ➔ 18열 학습기록 이관 & 안전백업) ----------
function migrateMemberAnalysisToLearningRecords() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ui = SpreadsheetApp.getUi();
  var sheetMember = getMemberAnalysisSheet(ss);
  if (!sheetMember) {
    ui.alert("[회원 분석] 시트를 찾을 수 없습니다.");
    return;
  }

  var resp = ui.alert(
    "회원 분석 시트 다이어트 & 학습기록 이관",
    "1) [회원 분석_백업] 시트를 먼저 안전하게 생성합니다.\n" +
    "2) F열(코칭 방향 & 주력 목표)에서 날짜가 찍힌 일일 수업메모 줄만 걸러내고 영구 코칭 메모는 그대로 보존합니다.\n" +
    "3) G열(이달의 학습 요약 전광판 & 최근 도서)의 과거 도서·진단점수 기록을 [학습기록] 18열 시트로 이관하고, G열에는 요약 전광판과 최근 도서 5권만 깔끔하게 남깁니다.\n" +
    "4) A~E열 및 H~P열은 단 한 글자도 건드리지 않습니다.\n\n진행할까요?",
    ui.ButtonSet.YES_NO
  );
  if (resp !== ui.Button.YES) return;

  // 1) 안전 백업 시트 생성
  var backupName = "회원 분석_백업";
  if (!ss.getSheetByName(backupName)) {
    var copied = sheetMember.copyTo(ss);
    copied.setName(backupName);
  }

  var sheetRec = vocabEnsureRecordSheet_(ss);
  sheetMember.getRange(1, 6).setValue("코칭 방향 & 주력 목표");
  sheetMember.getRange(1, 7).setValue("이달의 학습 요약 전광판 & 최근 도서");

  var lastRow = sheetMember.getLastRow();
  if (lastRow < 2) {
    ui.alert("회원 분석 시트에 학생 데이터가 없습니다.");
    return;
  }

  var data = sheetMember.getRange(2, 1, lastRow - 1, 7).getValues();
  var curYear = new Date().getFullYear();
  var migratedBookCount = 0;
  var cleanedStudentCount = 0;

  for (var i = 0; i < data.length; i++) {
    var rowNum = 2 + i;
    var name = String(data[i][1] || "").replace(/\s+/g, "");
    if (!name) continue;

    var rawF = String(data[i][5] || "").trim();
    var rawG = String(data[i][6] || "").trim();
    var recordsToMigrate = [];

    // F열 정리: 날짜 프리픽스(예: 26.4/15, 4/15)로 시작하는 일일 로그 줄만 분리하고 영구 코칭 방향은 보존
    if (rawF) {
      var fLines = rawF.split(/\r?\n/);
      var keptFLines = [];
      for (var fIdx = 0; fIdx < fLines.length; fIdx++) {
        var fl = fLines[fIdx].trim();
        if (!fl) continue;
        var fDateMatch = fl.match(/^(?:(\d{2})[\.\/])?(\d{1,2})\/(\d{1,2})\s+(.*)$/);
        if (fDateMatch) {
          // 날짜가 찍힌 과거 수업내용 줄 -> 학습기록 성실/문해력 발자국으로 보존 가능
          var yy = fDateMatch[1] ? (2000 + parseInt(fDateMatch[1], 10)) : curYear;
          var mm = ("0" + parseInt(fDateMatch[2], 10)).slice(-2);
          var dd = ("0" + parseInt(fDateMatch[3], 10)).slice(-2);
          var memoBody = String(fDateMatch[4] || "").trim();
          if (memoBody) {
            recordsToMigrate.push({
              date: yy + "-" + mm + "-" + dd,
              type: "일일요약",
              hwStatus: memoBody.indexOf("숙제완료") !== -1 ? "숙제완료" : (memoBody.indexOf("미지참") !== -1 ? "미지참" : ""),
              literacyMemo: memoBody
            });
          }
        } else {
          keptFLines.push(fl);
        }
      }
      var nextF = keptFLines.join("\n");
      if (nextF !== rawF) {
        sheetMember.getRange(rowNum, 6).setValue(nextF);
      }
    }

    // G열 정리: 날짜+책이름+진단점수 줄을 파싱해 [학습기록]에 이관하고 최근 5권 + 요약 전광판만 유지
    if (rawG) {
      var gLines = rawG.split(/\r?\n/);
      var bookLinesOnly = [];
      for (var gIdx = 0; gIdx < gLines.length; gIdx++) {
        var gl = gLines[gIdx].trim();
        if (!gl || gl.indexOf("📊") === 0) continue;
        bookLinesOnly.push(gl);
        var gDateMatch = gl.match(/^(?:(\d{2})[\.\/])?(\d{1,2})\/(\d{1,2})\s+(.+)$/);
        if (gDateMatch) {
          var gYy = gDateMatch[1] ? (2000 + parseInt(gDateMatch[1], 10)) : curYear;
          var gMm = ("0" + parseInt(gDateMatch[2], 10)).slice(-2);
          var gDd = ("0" + parseInt(gDateMatch[3], 10)).slice(-2);
          var rest = String(gDateMatch[4] || "").replace(/글완|글쓰기완료|글쓰기완|완료/g, "").trim();
          var scoreMatch = rest.match(/^(.*?)\s+(\d{2,3})\s*점?$/);
          var bTitle = scoreMatch ? scoreMatch[1].trim() : rest;
          var bScore = scoreMatch ? parseInt(scoreMatch[2], 10) : "";
          if (bTitle) {
            recordsToMigrate.push({
              date: gYy + "-" + gMm + "-" + gDd,
              type: "진단",
              book: bTitle,
              score: bScore
            });
            migratedBookCount++;
          }
        }
      }

      if (recordsToMigrate.length) {
        vocabUpsertStudentDayRecords_(sheetRec, name, recordsToMigrate);
      }

      var summaryLine = buildMemberSummaryBoardLine_(ss, name);
      var recentBooks = bookLinesOnly.length > 5 ? bookLinesOnly.slice(bookLinesOnly.length - 5) : bookLinesOnly;
      if (summaryLine) recentBooks.unshift(summaryLine);
      var nextG = recentBooks.join("\n");
      var gCell = sheetMember.getRange(rowNum, 7);
      applyRichTextStyling(gCell, nextG);
      cleanedStudentCount++;
    } else if (recordsToMigrate.length) {
      vocabUpsertStudentDayRecords_(sheetRec, name, recordsToMigrate);
      refreshMemberAnalysisSummaryBoardForStudent_(ss, name);
      cleanedStudentCount++;
    }
  }

  ui.alert(
    "✅ 회원 분석 다이어트 및 학습기록 이관 완료!\n\n" +
    "• 안전 백업 시트: [회원 분석_백업]\n" +
    "• 정리된 학생 수: " + cleanedStudentCount + "명\n" +
    "• [학습기록] 18열 시트로 이관된 도서·진단 기록: " + migratedBookCount + "건\n" +
    "• F열(코칭 방향 & 주력 목표)과 G열(이달의 학습 요약 전광판 & 최근 도서)이 날씬하게 정리되었습니다!"
  );
}

// ---------- 시트 셋업 (메뉴 8번) ----------
function vocabSetupSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ui = SpreadsheetApp.getUi();
  var msgs = [];

  var vs = ss.getSheetByName(VOCAB_SHEET);
  if (!vs) {
    vs = ss.insertSheet(VOCAB_SHEET);
    vs.getRange(1, 1, 1, VOCAB_HEADERS.length).setValues([VOCAB_HEADERS]).setFontWeight("bold").setBackground("#e6f4ea");

  } else {
    msgs.push("[어휘] 시트: " + Math.max(0, vs.getLastRow() - 1) + "개 어휘");
  }

  var rs = ss.getSheetByName(RECORD_SHEET);

// ---------- 오답 복습지 PDF (메뉴 9번: R열 틀린어휘 + [어휘] 시트 조합으로 문제지 즉석 생성!) ----------
var VOCAB_NUM = ["①", "②", "③", "④", "⑤"];
var VOCAB_MARK = ["㉠", "㉡", "㉢", "㉣", "㉤"];

function vocabShuffle_(arr) {
  var a = arr.slice();
  for (var i = a.length - 1; i > 0; i--) {
    var j = Math.floor(Math.random() * (i + 1));
    var t = a[i]; a[i] = a[j]; a[j] = t;
  }
  return a;
}

function vocabEsc_(s) {
  return String(s === undefined || s === null ? "" : s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}

function vocabLines_(txt, per) { return Math.max(1, Math.ceil(String(txt || "").length / per)); }

// [어휘] 시트 데이터로 틀린 어휘의 4지선다 복습 문제(빈칸/뜻맞히기) 동적 구성
function vocabBuildReviewQuestionFromBank_(wordStr, bookStr, levelStr, vocabRows) {
  var target = null;
  var normBook = vocabNormBook_(bookStr);
  for (var i = 0; i < vocabRows.length; i++) {
    var r = vocabRows[i];
    if (r.w === wordStr) {
      if (normBook && vocabNormBook_(r.book) === normBook) { target = r; break; }
      if (!target) target = r;
    }
  }
  if (!target) {
    return null;
  }
  // 오답 보기(distractors) 3개 추출 (같은 책 또는 같은 레벨 우선)
  var pool = [];
  for (var j = 0; j < vocabRows.length; j++) {
    var cand = vocabRows[j];
    if (cand.w === target.w || !cand.m) continue;
    if (cand.level === target.level || vocabNormBook_(cand.book) === normBook) {
      pool.push(cand);
    }
  }
  if (pool.length < 3) {
    for (var k = 0; k < vocabRows.length && pool.length < 10; k++) {
      if (vocabRows[k].w !== target.w && vocabRows[k].m) pool.push(vocabRows[k]);
    }
  }
  var shuffledPool = vocabShuffle_(pool);
  var label = (target.level ? target.level + " · " : "") + (target.book || "");

  // 예문이 있고 어휘(또는 활용형)가 예문에 포함되어 있으면 빈칸 문제, 아니면 뜻 고르기 문제 생성
  var formToBlank = target.form && target.ex.indexOf(target.form) !== -1 ? target.form : (target.ex.indexOf(target.w) !== -1 ? target.w : "");
  if (target.ex && formToBlank) {
    var wordOpts = [target.w];
    for (var d = 0; d < shuffledPool.length && wordOpts.length < 4; d++) {
      if (wordOpts.indexOf(shuffledPool[d].w) === -1) wordOpts.push(shuffledPool[d].w);
    }
    return {
      key: "w||" + target.w + "||" + (target.book || ""),
      type: "blank",
      word: target.w,
      meaning: target.m,
      label: label,
      sentence: target.ex.replace(formToBlank, "( )"),
      options: wordOpts,
      answer: target.w
    };
  }

  var meanOpts = [target.m];
  for (var mIdx = 0; mIdx < shuffledPool.length && meanOpts.length < 4; mIdx++) {
    if (meanOpts.indexOf(shuffledPool[mIdx].m) === -1) meanOpts.push(shuffledPool[mIdx].m);
  }
  return {
    key: "m||" + target.w + "||" + (target.book || ""),
    type: "mean",
    word: target.w,
    meaning: target.m,
    label: label,
    prompt: "다음 낱말의 뜻으로 알맞은 것은?",
    big: target.w,
    options: meanOpts,
    answer: target.m
  };
}

// 같은 문제(key)를 하나로 합치고 많이 틀린 순 → 최근순
function vocabConsolidate_(items) {
  var map = {}, order = [];
  for (var i = 0; i < items.length; i++) {
    var it = items[i];
    var k = it.rec.key;
    if (!map[k]) { map[k] = { key: k, rec: it.rec, count: 0, dates: [], last: 0 }; order.push(k); }
    var g = map[k];
    g.count++;
    g.dates.push(it.at);
    if (it.at >= g.last) { g.last = it.at; g.rec = it.rec; }
  }
  var out = [];
  for (var j = 0; j < order.length; j++) out.push(map[order[j]]);
  out.sort(function (a, b) { return (b.count - a.count) || (b.last - a.last); });
  return out;
}

// 출력용 한 문제 구성(보기 순서 섞기) + 대략의 줄 수(쪽 나누기 용)
function vocabPrepare_(g) {
  var r = g.rec, lines;
  if (r.type === "match") {
    var pairs = r.pairs || [];
    var right = vocabShuffle_(pairs.map(function (p) { return p.meaning; }));
    var ans = [];
    lines = 2 + pairs.length;
    for (var i = 0; i < right.length; i++) lines += vocabLines_(right[i], 24);
    for (var a = 0; a < pairs.length; a++) ans.push(right.indexOf(pairs[a].meaning));
    return { g: g, type: "match", left: pairs.map(function (p) { return p.word; }), right: right, pairs: pairs, matchAns: ans, lines: lines };
  }
  var opts = vocabShuffle_(r.options || []);
  var shortOpts = opts.every(function (o) { return String(o).length <= 8; });
  lines = 1 + (r.type === "mean" ? 0 : vocabLines_(r.type === "blank" ? r.sentence : r.big, 24));
  if (shortOpts) lines += 1; else for (var k = 0; k < opts.length; k++) lines += vocabLines_(opts[k], 24);
  return { g: g, type: r.type, options: opts, ansIndex: opts.indexOf(r.answer), short: shortOpts, lines: lines + 1 };
}

function vocabPaginate_(prepared, maxPerSheet, colCap) {
  var sheets = [], cur = null;
  function fresh() { cur = { cols: [[], []], used: [0, 0], count: 0 }; sheets.push(cur); }
  for (var i = 0; i < prepared.length; i++) {
    var p = prepared[i];
    if (!cur) fresh();
    var placed = false;
    if (cur.count < maxPerSheet) {
      for (var c = 0; c < 2 && !placed; c++) {
        if (cur.used[c] + p.lines <= colCap || (cur.count === 0 && c === 0)) {
          cur.cols[c].push(p); cur.used[c] += p.lines; cur.count++; placed = true;
        }
      }
    }
    if (!placed) { fresh(); cur.cols[0].push(p); cur.used[0] += p.lines; cur.count++; }
  }
  return sheets;
}

  var r = p.g.rec;
  var src = '<span class="src">' + vocabEsc_(r.label) + "</span>";

  if (p.type === "match") {
    h += "어휘에 알맞은 뜻의 번호를 쓰세요." + src;

    return h + "</div>";
  }
  if (p.type === "blank") h += "빈칸에 알맞은 말은?" + src + '<div class="o">' + vocabEsc_(r.sentence).replace("( )", '<span class="blank">&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;&nbsp;</span>') + "</div>";
  else if (p.type === "word") h += vocabEsc_(r.prompt) + src + '<div class="o">「' + vocabEsc_(r.big) + "」</div>";
  else h += vocabEsc_(r.prompt) + " <b>" + vocabEsc_(r.big) + "</b>" + src;
  if (p.short) {

    for (var a = 0; a < p.options.length; a++) parts.push(VOCAB_NUM[a] + " " + vocabEsc_(p.options[a]));
    h += '<div class="o">' + parts.join(" &nbsp; ") + "</div>";

    for (var b = 0; b < p.options.length; b++) h += '<div class="o">' + VOCAB_NUM[b] + " " + vocabEsc_(p.options[b]) + "</div>";
  }
  return h + "</div>";
}

  }
  return '<div class="a"><b>' + n + '.</b> <span class="ans">' + VOCAB_NUM[p.ansIndex] + " " + vocabEsc_(r.answer) + "</span>" + tag + "<br><b>" + vocabEsc_(r.word) + "</b>: " + vocabEsc_(r.meaning) + "</div>";

function vocabDateLabel_(times) {
  var seen = {}, out = [];
  for (var i = 0; i < times.length; i++) {

    var s = (d.getMonth() + 1) + "/" + d.getDate();
    if (!seen[s]) { seen[s] = true; out.push(s); }
  }
  return out.join(", ");
}

// 틀린어휘 문자열에서 '(짝짓기)' 또는 '짝짓기' 같은 문제유형 꼬리표 제거
function vocabCleanWrongWordsStr_(str) {
  if (!str) return "";
  var parts = String(str).split(",");
  var cleaned = [];
  for (var i = 0; i < parts.length; i++) {
    var w = String(parts[i] || "").trim();
    if (!w || w === "짝짓기" || w === "(짝짓기)") continue;
    if (cleaned.indexOf(w) === -1) cleaned.push(w);
  }
  return cleaned.join(", ");
}

function buildStudentDailyBriefingText_(name, dateStr, rows18) {

    var vTot = r[6], vCor = r[7], vRate = r[8], dScore = r[9];

    if (bk || vTot !== "" || dScore !== "") {

      var sub = [];
      if (dScore !== "" && dScore !== null && dScore !== undefined) sub.push("독서이해도 진단 " + dScore + "점");
      if (vTot !== "" && vTot !== null && vTot !== undefined) {
        sub.push("어휘학습 " + vTot + "문항 중 " + (vCor !== "" ? vCor : 0) + "개 정답(" + (vRate !== "" ? vRate : 0) + "%)");

      if (sub.length) bLine += " — " + sub.join(" / ");
      if (step2) bLine += "\n  └ 독서발자국: \"" + step2 + "\"";
      books.push(bLine);
    }
    if (wrong) wrongWordsAll.push(wrong);

    var trName = String(r[11] || "").trim();
    var trDet = String(r[12] || "").trim();
    var trErr = r[13];
    var trUnd = String(r[14] || "").trim();

  lines.push("안녕하세요, " + name + " 학부모님! 오늘(" + md + ") 학원에서 진행한 수업 활동 안내드립니다.");

    lines.push("✅ 숙제 현황: " + (hwStatus || "확인 완료") + (step1Memo ? " (" + step1Memo + ")" : ""));
  }
  if (books.length) {
    lines.push("📚 독서 & 어휘 학습");
    for (var b = 0; b < books.length; b++) lines.push(books[b]);
  }
  if (wrongWordsAll.length) {
    lines.push("📝 오늘 복습한 어휘: " + wrongWordsAll.join(", "));
  }
  if (trainings.length) {
    lines.push("✏️ 문해력 & 국어 트레이닝");
    for (var t = 0; t < trainings.length; t++) lines.push(trainings[t]);
  }
  if (reflection) {

  }
  lines.push("");
  lines.push("가정에서도 따뜻한 격려와 칭찬 부탁드립니다. 감사합니다! 😊");
  return lines.join("\n");
